package guestbridge

import (
    "encoding/json"
    "os"
    "testing"
)

func TestVZ3MaximumViewUnions(t *testing.T) {
    raw, err := os.ReadFile(os.Getenv("HFS_VIEW_BOUNDS"))
    if err != nil { t.Fatal(err) }
    var values struct { Commands map[string]json.RawMessage; Replies map[string]json.RawMessage }
    if err = json.Unmarshal(raw, &values); err != nil { t.Fatal(err) }
    for op, command := range values.Commands {
        c, err := DecodeCommandVersion(command, 1)
        if err != nil { t.Fatalf("%s command rejected: %v", op, err) }
        if response, ok := values.Replies[op]; ok {
            var envelope struct { Payload json.RawMessage }
            if err = json.Unmarshal(response, &envelope); err != nil { t.Fatal(err) }
            if err = validateViewResponse(c, envelope.Payload); err != nil { t.Fatalf("%s reply rejected: %v", op, err) }
        }
    }
}
