type receivedCarrier struct {

func startReceivedCarrier(t *testing.T, r *sourceRig) *receivedCarrier {
	return startReceivedCarrierMode(t, r, false)
}
func startReceivedCarrierMode(t *testing.T, r *sourceRig, backendOnly bool) *receivedCarrier {
	return startReceivedCarrierBudget(t, r, backendOnly, GuestArtifactBudget{OrdinaryBytes: 256 << 20, ControlBytes: 256 << 20, Reservations: 8192, Cookies: 4096, Received: 4096})
}
func startReceivedCarrierBudget(t *testing.T, r *sourceRig, backendOnly bool, budget GuestArtifactBudget) *receivedCarrier {
	t.Helper()
	// Both paths are explicit: a materialized test must fail on absent inputs.
	build, source := os.Getenv("HFS_C6_BUILD"), os.Getenv("HFS_C6_SOURCE")
	if build == "" || source == "" {
		t.Fatal("HFS_C6_BUILD and HFS_C6_SOURCE required for actual C6 consumer")
	}
	group := GuestProvisionGroup{Authorization: r.artifactAuthorization}
	credentials := make([]guestbridge.Credentials, len(r.peers))
	contexts := []guestbridge.Context{}
	for i, peer := range r.peers {
		credentials[i] = peer.credentials
		if i < 16 { contexts = append(contexts, peer.scope) }
		group.Signers = append(group.Signers, GuestArtifactSigners{Host: peer.key, Guest: r.artifactGuest})
		group.LocalMounts = append(group.LocalMounts, GuestArtifactMount{DomainIndex: i, Session: peer.scope.Session, DeviceTag: fmt.Sprintf("hfs%d", i), DeviceName: fmt.Sprintf("virtio%d", i), UserNamespace: peer.credentials.UserNamespace, IDMapHash: peer.credentials.IDMapHash, Realm: testUUID(100).String(), Subject: "616263", Anchor: testUUID(101).String(), Nonce: testUUID(103).String(), ViewControlNonce: r.artifactAuthorization.Enrollment.TicketFamilies[i].Control.Nonce})
	}
	provision, err := SignGuestProvision(r.scope.GuestBoot, r.installed.Admission.ProviderNonce, []GuestProvisionGroup{group}, budget, r.artifactOperator)
	require(t, err)
	bind, err := SignGuestBind(provision, group.Authorization.Enrollment.ID.String(), r.supervisor.record.Controller.String(), r.artifactOperator, group.Signers)
	require(t, err)
	output := filepath.Join(r.temporary(), "build")
	compiler := exec.CommandContext(t.Context(), "python3", build, "--source", source, "--output", output, "--operator-key", hex.EncodeToString(r.artifactOperator.Public().(ed25519.PublicKey)))
	log, err := compiler.CombinedOutput()
	if err != nil {
		t.Fatalf("actual C6 build: %v\n%s", err, log)
	}
	consumer := filepath.Join(output, "consumer")
	if _, err := os.Stat(consumer); err != nil {
		t.Fatalf("missing actual C6 consumer: %v", err)
	}
	c := &receivedCarrier{t: t, rig: r, backendOnly: backendOnly, commands: map[uint64]guestbridge.Command{}, originals: map[string][]byte{}, responses: map[uint64][]byte{}}
	c.cmd = exec.CommandContext(t.Context(), consumer)
	ledger := r.temporary()
	require(t, os.Chmod(ledger, 0700))
	c.cmd.Env = append(os.Environ(), "HFS_C6_LEDGER="+ledger, "ASAN_OPTIONS=detect_leaks=0:allocator_may_return_null=1:quarantine_size_mb=16")
	if backendOnly {
		c.cmd.Env = append(c.cmd.Env, "HFS_C6_BACKEND_ONLY=1")
	}
	c.input, err = c.cmd.StdinPipe()
	require(t, err)
	stdout, err := c.cmd.StdoutPipe()
	require(t, err)
	c.reader = bufio.NewScanner(stdout)
	c.reader.Buffer(make([]byte, 65536), 4*guestbridge.BodyBytes+256)
	c.cmd.Stderr = &c.diagnostics
	require(t, c.cmd.Start())
	t.Cleanup(func() {
		if c.cmd.ProcessState == nil {
			_ = c.cmd.Process.Kill()
			_ = c.cmd.Wait()
		}
	})
	c.write(string(mustRuntimeJSON(t, map[string]any{"base": *r.mappingNativeCell, "contexts": contexts, "protocol_features": guestbridge.TerminalProtocol().Features, "binds": []json.RawMessage{bind}, "credentials": credentials, "os": map[string]any{"controller": r.supervisor.record.Controller.String(), "native_endpoint": r.native.Endpoint, "boot_id": r.scope.GuestBoot, "provider_nonce": r.installed.Admission.ProviderNonce, "capture_tuples": r.credentialCaptures}, "provision": json.RawMessage(provision)})))
	c.readUntil(0, true)
	return c
}
func (d *receivedDiagnostics) Write(raw []byte) (int, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	return d.buffer.Write(raw)
}
func (d *receivedDiagnostics) String() string {
	d.mu.Lock()
	defer d.mu.Unlock()
	return d.buffer.String()
}

func (c *receivedCarrier) write(line string) {
	c.t.Helper()
	_, err := fmt.Fprintln(c.input, line)
	require(c.t, err)
}
func (c *receivedCarrier) number(s string) uint64 {
	c.t.Helper()
	n, err := strconv.ParseUint(s, 10, 64)
	require(c.t, err)
	return n
}
func (c *receivedCarrier) readUntil(id uint64, startup bool) (int, []byte) {
	c.t.Helper()
	for c.reader.Scan() {
		fields := strings.Fields(c.reader.Text())
		if len(fields) == 1 && fields[0] == "READY" && startup {
			return 0, nil
		}
		if len(fields) < 2 {
			c.t.Fatalf("invalid C6 IPC %q", c.reader.Text())
		}
		switch fields[0] {
		case "L":
			c.t.Log(strings.Join(fields[1:], " "))
		case "S":
			if c.backendOnly {
				continue
			} // This mode explicitly substitutes the OS observer.
			if len(fields) != 2 {
				c.t.Fatal("snapshot arity")
			}
			raw, err := hex.DecodeString(fields[1])
			require(c.t, err)
			var admission kernelAdmission
			require(c.t, strictJSON(raw, &admission))
			// Replace only the complete actual serializer record; static observations
			// retain their independently measured manifest, notes, boot and ABI fields.
			c.rig.installed.Admission = admission
			next := c.rig.observationFile + ".next"
			require(c.t, os.WriteFile(next, mustRuntimeJSON(c.t, c.rig.installed), 0600))
			require(c.t, os.Rename(next, c.rig.observationFile))
		case "W":
			c.nativeWrite(fields)
		case "R":
			c.nativeRead(fields)
		case "Y":
			yielded := c.number(fields[1])
			if yielded != id && !startup {
				c.t.Fatalf("unexpected scheduling ID %d want %d", yielded, id)
			}
			if c.yields != nil {
				c.yields(yielded)
			}
			c.write("CONTINUE")
		case "D":
			if startup || len(fields) != 4 || c.number(fields[1]) != id {
				c.t.Fatalf("unexpected completion %q", c.reader.Text())
			}
			code, err := strconv.Atoi(fields[2])
			require(c.t, err)
			if fields[3] == "-" {
				return code, nil
			}
			raw, err := hex.DecodeString(fields[3])
			require(c.t, err)
			return code, raw
		default:
			c.t.Fatalf("unexpected IPC %q", c.reader.Text())
		}
	}
	c.t.Fatalf("C6 ended before completion: %v %s", c.reader.Err(), c.diagnostics.String())
	return -1, nil
}
func (c *receivedCarrier) nativeWrite(fields []string) {
	c.t.Helper()
	if len(fields) != 3 {
		c.t.Fatal("native write arity")
	}
	offset := c.number(fields[1])
	raw, err := hex.DecodeString(fields[2])
	require(c.t, err)
	frame, err := guestbridge.DecodeFrame(raw)
	require(c.t, err)
	if frame.Offset != 0 || uint32(len(frame.Data)) != frame.Total {
		c.t.Fatal("unexpected fragmented C6 command")
	}
	command, err := guestbridge.DecodeCommandVersion(frame.Data, 4)
	require(c.t, err)
	if command.Delegation == nil {
		c.t.Fatal("C6 native request omitted actual delegation")
	}
	d := command.Delegation
	original := command
	original.Cell = d.GuestCell
	original.Op = d.GuestOp
	original.Reclaim = d.GuestReclaim
	original.Delegation = nil
	if original.Op == "RELEASE_CELL" {
		original.Payload = mustRuntimeJSON(c.t, d.GuestReclaim)
	}
	key := fmt.Sprintf("%s/%d", command.Context.Session, d.GuestCell)
	originalBytes := mustRuntimeJSON(c.t, original)
	if previous := c.originals[key]; previous != nil && !bytes.Equal(previous, originalBytes) {
		c.t.Fatal("C6 continuation changed full original credentials/payload/Ack")
	}
	c.originals[key] = originalBytes
	if old, ok := c.commands[frame.Cell]; ok && !bytes.Equal(mustRuntimeJSON(c.t, old), frame.Data) {
		c.t.Fatal("actual native retry changed immutable request bytes")
	}
	c.commands[frame.Cell] = command
	n, err := c.rig.mappingTransport.Write(c.t.Context(), c.rig.native, c.rig.native.Endpoint.CommandNode, offset, raw)
	if err != nil {
		c.write("ERROR")
	} else {
		c.write(fmt.Sprintf("OK %d", n))
	}
}
func (c *receivedCarrier) nativeRead(fields []string) {
	c.t.Helper()
	if len(fields) != 4 {
		c.t.Fatal("native read arity")
	}
	offset, count := c.number(fields[2]), c.number(fields[3])
	if c.holdNativeResponses {
		// Declared OS delivery delay only; no response or receipt is fabricated.
		c.nativeAgain++
		c.write("AGAIN")
		return
	}
	node := c.rig.native.Endpoint.ResultNode
	if fields[1] == "12" {
		node = c.rig.native.Endpoint.RecallNode
	} else if fields[1] != "11" {
		c.t.Fatal("unknown native fd")
	}
	raw, err := c.rig.mappingTransport.Read(c.t.Context(), c.rig.native, node, offset, uint32(count))
	if errors.Is(err, fsapi.WouldBlock) {
		c.nativeAgain++
		c.write("AGAIN")
		return
	}
	require(c.t, err)
	if offset%guestbridge.CellStride == guestbridge.HeaderBytes {
		c.responses[offset/guestbridge.CellStride] = bytes.Clone(raw)
	}
	c.write("DATA " + hex.EncodeToString(raw))
}
func (c *receivedCarrier) action(verb string, session int, tail string, want int) []byte {
	c.t.Helper()
	c.nextID++
	id := c.nextID
	c.write(strings.TrimSpace(fmt.Sprintf("%s %d %d %s", verb, id, session, tail)))
	code, raw := c.readUntil(id, false)
	if code != want {
		for cell, command := range c.commands {
			if command.Op == "GAP_RESOLVE" || command.Op == "QUERY" || command.Op == "RETIRE" {
				c.t.Logf("actual retained exchange cell%d %s request=%s response=%s", cell, command.Op, command.Payload, c.responses[cell])
			}
		}
		c.t.Fatalf("C6 %s session%d %s = %d want%d body=%s diagnostics=%s", verb, session, tail, code, want, raw, c.diagnostics.String())
	}
	return raw
}
func (c *receivedCarrier) run(session int, reserved bool, op string, payload any) []byte {
	c.t.Helper()
	lane := 0
	if reserved {
		lane = 1
	}
	return c.action("RUN", session, fmt.Sprintf("%d %s %s", lane, op, hex.EncodeToString(mustRuntimeJSON(c.t, payload))), 0)
}
func (c *receivedCarrier) call(session int, op string, args any) []byte {
	c.t.Helper()
	peer := c.rig.peers[session]
	peer.claims++
	intent := guestbridge.Intent{Opcode: op, Args: mustRuntimeJSON(c.t, args)}
	call := guestbridge.Call{Claim: guestbridge.Claim{Realm: testUUID(100).String(), Subject: "616263", Anchor: testUUID(101).String(), Nonce: testUUID(103).String(), Number: peer.claims, OriginalHash: hashBytes(mustRuntimeJSON(c.t, intent))}, Credentials: peer.credentials, Intent: intent}
	c.run(session, false, "ALLOCATE", call)
	c.run(session, false, "CALL", call)
	return c.run(session, true, "QUERY", map[string]any{"claim": call.Claim})
}
func (c *receivedCarrier) close() {
	c.t.Helper()
	c.write("END")
	require(c.t, c.input.Close())
	require(c.t, c.cmd.Wait())
}

func TestControlledReceivedActualProviderProducer(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	for i := range 2 {
		c.action("RELEASE_NEGATIVES", i, "", 0)
	}
	domains := make([]guestbridge.Capability, 2)
	for i := range 2 {
		root := c.call(i, "ROOT", map[string]any{})
		caps := receivedResult(t, root)
		require(t, json.Unmarshal(c.action("ADOPT_ROOT", i, "", 0), &domains[i]))
		created := c.call(i, "CREATE", map[string]any{"parent": caps[0], "name": fmt.Sprintf("66696c65%02x", i+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}})
		receivedResult(t, created)
		c.action("ADOPT_DESCRIPTION", i, "CREATE", 0)
	}
	for i := range 2 {
		c.action("ACQUIRE", i, "", 0)
		for _, check := range []string{"target", "publication", "cap", "cookie_balance"} {
			c.action("CHECK", i, check, 0)
		}
		for _, check := range []string{"forged_target", "forged_event", "forged_publication", "nonnull_publication", "wrong_channel", "bad_device", "bad_map", "retirement_refuses"} {
			c.action("CHECK", i, check, 0)
		}
	}
	empty := c.run(0, true, "VIEW_WAIT", map[string]any{"domain": domains[0], "after_sequence": 0})
	var initial struct{ Payload struct{ Status string } }
	require(t, json.Unmarshal(empty, &initial))
	if initial.Payload.Status != "EMPTY" {
		t.Fatalf("initial actual WAIT must expire EMPTY: %s", empty)
	}
	before := c.nativeAgain
	waitBase := uint64(0)
	for cell := range c.commands {
		if cell > waitBase {
			waitBase = cell
		}
	}
	nested := false
	c.yields = func(_ uint64) {
		continuation := false
		for cell, command := range c.commands {
			if cell > waitBase && command.Context == r.scope && command.Delegation.GuestOp == "VIEW_WAIT" && command.Delegation.PollIndex >= 1 {
				continuation = true
			}
		}
		if !nested && continuation {
			nested = true
			c.yields = nil
			c.action("RELEASE_START", 1, "", 0)
		}
	}
	event := c.run(0, true, "VIEW_WAIT", map[string]any{"domain": domains[0], "after_sequence": 0})
	c.yields = nil
	if !nested || c.nativeAgain <= before {
		t.Fatal("WAIT did not yield after actual backend EAGAIN")
	}
	verifyReceivedWait(t, c, 0, empty, event)
	c.action("CHECK", 0, "event", 0)
	c.action("CHECK", 0, "forged_event", 0)
	c.action("CHECK", 0, "wrong_channel", 0)
	for _, phase := range []string{"STOPPED", "INSTALLED_DRAIN", "DRAINED", "INSTALLED_FINAL", "REOPENED"} {
		for i := range 2 {
			if i != 0 || phase != "STOPPED" {
				c.run(i, true, "VIEW_WAIT", map[string]any{"domain": domains[i], "after_sequence": 0})
			}
			c.action("PHASE", i, phase, 0)
		}
	}
	for i := range 2 {
		c.action("CHECK", i, "stale_target", 0)
		c.action("CHECK", i, "cap", 0)
		c.action("CHECK", i, "cookie_balance", 0)
	}
	c.close()
}
func verifyReceivedWait(t *testing.T, c *receivedCarrier, session int, previous, response []byte) {
	t.Helper()
	var reply struct {
		Cell         uint64
		Verification *guestbridge.VerificationReceiptV2
	}
	require(t, json.Unmarshal(response, &reply))
	if reply.Verification == nil || reply.Verification.Kind != "EVENT" || reply.Verification.PollCount < 2 {
		t.Fatalf("actual multipoll EVENT receipt missing: %s", response)
	}
	key := fmt.Sprintf("%s/%d", c.rig.peers[session].scope.Session, reply.Cell)
	original := c.originals[key]
	_, err := guestbridge.VerifyMappedReceipt(c.rig.peers[session].mappedBinding(), original, response)
	require(t, err)
	var command guestbridge.Command
	require(t, json.Unmarshal(original, &command))
	var prior struct{ Cell uint64 }
	require(t, json.Unmarshal(previous, &prior))
	if command.Reclaim == nil || command.Reclaim.Cell != prior.Cell || command.Reclaim.ResponseHash != hashBytes(previous) {
		t.Fatal("WAIT did not retain exact prior EMPTY response Ack")
	}
	legs := 0
	for _, native := range c.commands {
		d := native.Delegation
		if native.Context.Session != command.Context.Session || d.GuestCell != command.Cell {
			continue
		}
		legs++
		if d.GuestReclaim == nil || *d.GuestReclaim != *command.Reclaim {
			t.Fatal("native poll changed original Ack")
		}
	}
	if legs != int(reply.Verification.PollCount) {
		t.Fatalf("real poll count=%d signed=%d", legs, reply.Verification.PollCount)
	}
	var native, guest map[string]json.RawMessage
	require(t, json.Unmarshal(c.responses[reply.Verification.Cell], &native))
	require(t, json.Unmarshal(response, &guest))
	if !bytes.Equal(native["verification"], guest["verification"]) {
		t.Fatal("C carrier changed Producer signature bytes")
	}
}

func TestControlledReceivedActualSourceCorruption(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	c.call(0, "ROOT", map[string]any{})
	c.action("ADOPT_ROOT", 0, "", 0)
	// Establish an actual retained source before changing its actual journal bytes.
	c.action("ACQUIRE", 0, "", 0)
	before := uint64(0)
	for cell := range c.commands {
		if cell > before {
			before = cell
		}
	}
	corruptReceivedJournal(t, r)
	// The real Producer withholds the damaged SOURCE: the actual native worker
	// publishes no response, and the actual C exchange reaches its own deadline.
	c.action("ACQUIRE", 0, "", -110)
	attempts := 0
	for cell, command := range c.commands {
		if cell <= before {
			continue
		}
		if command.Op == "VIEW_ACQUIRE" {
			attempts++
			t.Logf("corrupted-source actual VIEW_ACQUIRE native cell %d: no response=%t", cell, c.responses[cell] == nil)
			if c.responses[cell] != nil {
				t.Fatal("damaged source minted an actual response")
			}
		}
	}
	if attempts != 1 {
		t.Fatalf("corruption must reach exactly one real new source RPC: %d", attempts)
	}
	c.action("CHECK", 0, "cookie_balance", 0)
	c.close()
}
func corruptReceivedJournal(t *testing.T, r *sourceRig) {
	t.Helper()
	stream := r.artifactAuthorization.Enrollment.Domains[0].Journal.Stream
	recovered, err := segment.Recover(t.Context(), r.source, stream, segment.RecoveryOptions{})
	require(t, err)
	if len(recovered.Segments) == 0 {
		t.Fatal("no actual source journal segment")
	}
	f, err := r.source.Open(t.Context(), recovered.Segments[0].Name, os.O_RDWR, 0)
	require(t, err)
	defer func() { require(t, f.Close()) }()
	var original [1]byte
	n, err := f.ReadAt(t.Context(), original[:], 32)
	require(t, err)
	if n != 1 {
		t.Fatal("short actual journal read")
	}
	changed := original
	changed[0] ^= 1
	n, err = f.WriteAt(t.Context(), changed[:], 32)
	require(t, err)
	if n != 1 {
		t.Fatal("short actual journal corruption")
	}
	require(t, f.Sync(t.Context()))
	var actual [1]byte
	n, err = f.ReadAt(t.Context(), actual[:], 32)
	require(t, err)
	if n != 1 || actual != changed || actual == original {
		t.Fatal("actual journal bytes did not change")
	}
	t.Logf("actual source journal %s offset32 byte %02x -> %02x (synced and reread)", recovered.Segments[0].Name, original[0], actual[0])
}

func receivedResult(t *testing.T, raw []byte) []guestbridge.Capability {
	t.Helper()
	var reply struct {
		Payload struct {
			Status string
			Result struct {
				Code         string
				Capabilities []guestbridge.Capability
				Publication  json.RawMessage
			}
		}
	}
	require(t, json.Unmarshal(raw, &reply))
	if reply.Payload.Status != "TERMINAL" || reply.Payload.Result.Code != "OK" || len(reply.Payload.Result.Capabilities) != 2 || !bytes.Equal(reply.Payload.Result.Publication, []byte("null")) {
		t.Fatalf("actual terminal result/capabilities/null publication: %s", raw)
	}
	return reply.Payload.Result.Capabilities
}

func runControlledReceivedFundedDirty(t *testing.T, retainedPressure bool) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	domains := make([]guestbridge.Capability, 2)
	request := func(i int, op string, args any) []byte {
		return c.action("REQUEST", i, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
	}
	for i := range 2 {
		caps := receivedResult(t, request(i, "ROOT", map[string]any{}))
		require(t, json.Unmarshal(c.action("ADOPT_ROOT_PREPARE", i, "", 0), &domains[i]))
		receivedResult(t, request(i, "CREATE", map[string]any{"parent": caps[0], "name": fmt.Sprintf("6469727479%02x", i+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
		c.action("ADOPT_DESCRIPTION", i, "CREATE", 0)
		c.action("VIEW_NEW", i, "", 0)
		c.action("VIEW_NEW", i, "", -17)
	}
	c.action("DIRTY", 0, "", 0)
	c.action("ACQUIRE_PAGE", 1, "", 0)
	if retainedPressure {
		for i := range 2 {
			c.action("OWNER_NEGATIVES", i, "", 0)
			c.action("VIEW_QUERY_HOLD", i, "", 0)
		}
		full := false
		for round := 0; round < 32 && !full; round++ {
			for i := range 2 {
				var pressure struct {
					Held      int
					Exhausted bool

			c.action("PHASE", i, "STOPPED", -16)
			c.action("PIN", i, "0", 0)
		}
		c.action("PHASE", i, "STOPPED", 0)

		}

			c.action("RELEASE_DONE", i, "", 0)

			c.action("DROP_PRESSURE", i, "", 0)
			c.action("RETIRE", i, "", 0)
		}
	}

		c.action("RELEASE_DONE", i, "", 0)
		c.action("RETIRE", i, "", 0)

			c.action("VIEW_RETIRE", i, "", 0)
			continue
		}

		// comparison allocation fails. A later acquisition cannot replace it.
		c.action("FAULT", i, "VIEW_RETIRE 3", 0)

			for j := range 2 {
				c.run(j, true, "VIEW_WAIT", map[string]any{"domain": domains[j], "after_sequence": 0})

			}
		}

				}

		}
	}

	}

// Reconstruct only the documented scalar translation of the recorded actual
// native receipt. This is an inert cross-language check; the real guest callback
// has already consumed the actual privately received response before this runs.
func checkReceivedViewFloorBindings(t *testing.T, c *receivedCarrier, extended bool) {
	t.Helper()
	count := 0
	blocked := 0
	var blockedKey string
	floors := make(map[string]struct {
		Through uint64
		Hashes  []string `json:"result_hashes"`
	})
	for cell, command := range c.commands {
		if command.Op != "VIEW_RETIRE" {
			continue
		}
		var fields map[string]json.RawMessage
		require(t, json.Unmarshal(c.responses[cell], &fields))
		var cut struct {
			Anchor, Nonce string
			Credentials   guestbridge.Credentials
			Through       uint64
			Hashes        []string `json:"result_hashes"`
		}
		require(t, json.Unmarshal(command.Payload, &cut))
		key := command.Context.Session + "/" + cut.Anchor + "/" + cut.Nonce
		var response struct {
			Payload struct {
				Status string
				Floor  uint64
				Proof  *string `json:"proof_hash"`
			}
		}
		require(t, json.Unmarshal(c.responses[cell], &response))
		var receipt guestbridge.VerificationReceiptV2
		require(t, json.Unmarshal(fields["verification"], &receipt))
		if receipt.Kind != "VIEW_FLOOR" {
			if response.Payload.Status != "BLOCKED" || response.Payload.Floor != 0 || response.Payload.Proof != nil ||
				command.Context != c.rig.peers[1].scope || cut.Through != 1 || len(cut.Hashes) != 1 || !bytes.Equal(mustRuntimeJSON(t, cut.Credentials), mustRuntimeJSON(t, c.rig.credentials)) {
				t.Fatalf("unexpected actual blocked view cut: %s / %s", command.Payload, c.responses[cell])
			}
			blocked++
			blockedKey = key
			continue
		}
		floors[key] = struct {
			Through uint64
			Hashes  []string `json:"result_hashes"`
		}{cut.Through, cut.Hashes}
		count++
		fields["cell"] = mustRuntimeJSON(t, receipt.GuestCell)
		fields["op"] = mustRuntimeJSON(t, "VIEW_RETIRE")
		original := c.originals[fmt.Sprintf("%s/%d", command.Context.Session, receipt.GuestCell)]
		if len(original) == 0 {
			t.Fatal("missing actual original VIEW_RETIRE")
		}
		binding := c.rig.peers[0].mappedBinding()
		for _, peer := range c.rig.peers {
			if peer.scope == command.Context {
				binding = peer.mappedBinding()
			}
		}
		_, err := guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields))
		require(t, err)
		changed := receipt
		changed.MappingHash = hashBytes([]byte("wrong mapping"))
		fields["verification"] = mustRuntimeJSON(t, changed)
		if _, err = guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted changed mapping")
		}
		changed = receipt
		changed.PollCount = 2
		fields["verification"] = mustRuntimeJSON(t, changed)
		if _, err = guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted wrong poll mapping")
		}
		fields["verification"] = mustRuntimeJSON(t, receipt)
		wrongBinding := binding
		wrongBinding.Controller = testUUID(120).String()
		if _, err = guestbridge.VerifyMappedReceipt(wrongBinding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted other controller")
		}
	}
	if !extended {
		if count != 4 || blocked != 0 {
			t.Fatalf("pressure view floors/blocked %d/%d", count, blocked)
		}
		return
	}
	if count != 5 || blocked != 1 {
		t.Fatalf("actual signed view floors/blocked cuts %d/%d", count, blocked)
	}
	final := floors[blockedKey]
	if final.Through != 2 || len(final.Hashes) != 2 {
		t.Fatalf("final control floor omitted real releases: %+v", final)
	}
	actual := make(map[string]bool)
	for cell, command := range c.commands {
		if command.Op != "VIEW_RELEASE" || command.Context != c.rig.peers[1].scope {
			continue
		}
		var reply struct{ Payload struct{ Status, ID string } }
		require(t, json.Unmarshal(c.responses[cell], &reply))
		if reply.Payload.Status == "RELEASED" {
			actual[reply.Payload.ID] = true
		}
	}
	// Final release may be completed by VIEW_QUERY; its protected payload
	// retains the original VIEW_RELEASE operation and exact response self-id.
	for cell, command := range c.commands {
		if command.Op != "VIEW_QUERY" || command.Context != c.rig.peers[1].scope {
			continue
		}
		var reply struct {
			Payload struct{ Status, ID, Operation string }
		}
		require(t, json.Unmarshal(c.responses[cell], &reply))
		if reply.Payload.Status == "RELEASED" && reply.Payload.Operation == "VIEW_RELEASE" {
			actual[reply.Payload.ID] = true
		}
	}
	if len(actual) != 2 || !actual[final.Hashes[0]] || !actual[final.Hashes[1]] || final.Hashes[0] == final.Hashes[1] {
		t.Fatalf("control floor differs from actual release replies: %v / %v", final.Hashes, actual)
	}
}

func TestControlledReceivedActualTerminalDirtyActor(t *testing.T) {

	request := func(i int, op string, args any) []byte {
		return c.action("REQUEST", i, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
	}
	for i := range 1 {
		caps := receivedResult(t, request(i, "ROOT", map[string]any{}))
		c.action("ADOPT_ROOT_PREPARE", i, "", 0)
		receivedResult(t, request(i, "CREATE", map[string]any{"parent": caps[0], "name": fmt.Sprintf("6163746f72%02x", i+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
		c.action("ADOPT_DESCRIPTION", i, "CREATE", 0)
		// A lookup reference can be forgotten while its independent open
		// description remains live, before acquiring physical/dirty authority.
		c.action("PATH_FORGET", i, "", 0)
		// Settle completed setup claims before admitting dirty capacity. Live
		// capabilities keep their genuine independent Received owners.
		c.action("REPLIES_DROP", i, "", 0)
		c.action("RETIRE", i, "", 0)
		c.action("VIEW_NEW", i, "", 0)
	}
	for i := range 1 {
		c.action("ACTOR_START", i, "", 0)
	}
	c.action("DIRTY", 0, "", 0)
	// The final description release is already admitted when it waits for
	// recall. Schedule STOP and the real actors at its actual QUERY yield;
	// shutdown must retain the dirty request until those actors drain it.
	through := uint64(0)
	for cell := range c.commands {
		if cell > through {
			through = cell
		}
	}
	drained := false
	c.yields = func(_ uint64) {
		pending := false
		for cell, command := range c.commands {
			if cell > through && command.Context == r.scope && command.Delegation.GuestOp == "QUERY" && len(c.responses[cell]) != 0 {
				pending = true
			}
		}
		if !pending {
			return
		}
		// Use the complete production release path. Its initial private
		// response must release the family admission lock before freeze.
		c.yields = func(_ uint64) {
			var ready bool
			require(t, json.Unmarshal(c.action("RELEASE_READY", 0, "", 0), &ready))
			if !ready {
				return
			}
			c.yields = nil
			c.action("CLOSE_PENDING", 0, "", 0)

				for _, queue := range []string{"poll", "event", "drain", "event"} {
					c.action("ACTOR_STEP", 0, queue, 0)
				}

				require(t, json.Unmarshal(c.action("ACTOR_STATE", 0, "", 0), &state))
				drained = state.Phase == "REOPENED"
			}
			if !drained {
				t.Fatal("actual C control actors failed to finish recall")
			}
		}
		c.action("RELEASE_DONE", 0, "", 0)
		c.yields = nil
	}
	c.action("DESCRIPTION_CLOSE", 0, "", 0)
	c.yields = nil
	if !drained {
		t.Fatal("description release never yielded with actual pending QUERY")
	}
	for i := range 1 {
		c.action("RETIRE", i, "", 0)
		c.action("VIEW_RETIRE", i, "", 0)
		c.action("REPLIES_DROP", i, "", 0)
		c.action("ROOT_DROP", i, "", 0)
	}
	c.action("DESTROY", 0, "", 0)
	c.action("TOMBSTONE", 0, "", 0)
	receivedResult(t, c.action("REQUEST", 1, "ROOT "+hex.EncodeToString(mustRuntimeJSON(t, map[string]any{})), 0))

// Full 80-cell composition of the actual C exporter with the actual Go owners.
// This mode explicitly substitutes the OS/kernel observer, as the host-only
// owner tests do. It does not execute Linux teardown or mint private Received;
// the separate dirty test exercises those production-source lifetime paths.
func TestControlledReceivedActualTerminalBackend80(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.TerminalProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrierMode(t, r, true)
	exchange := func(session int, cell uint64, op string, payload any, ack *guestbridge.Ack) ([]byte, []byte, guestbridge.VerificationReceiptV2) {
		peer := r.peers[session]
		original := mustRuntimeJSON(t, map[string]any{"cell": cell, "context": peer.scope, "delegation": nil, "op": op, "payload": payload, "protocol": guestbridge.TerminalProtocol(), "reclaim": ack})
		response := c.action("BACKEND_EXCHANGE", session, hex.EncodeToString(original), 0)
		var receipt guestbridge.VerificationReceiptV2
		if op == "SESSION_RETIRE" {
			var err error
			receipt, err = guestbridge.VerifyMappedReceipt(peer.mappedBinding(), original, response)
			require(t, err)
		} else {
			// HELLO/blocked RESUME own transport replies, not source or terminal
			// authority: their existing closed contract has verification:null.
			var envelope struct {
				Cell         uint64
				Context      guestbridge.Context
				Op           string
				Protocol     guestbridge.Protocol
				Verification json.RawMessage
				Payload      struct{ Status string }
			}
			require(t, json.Unmarshal(response, &envelope))
			status := "ACCEPTED"
			if op == "RESUME" {
				status = "BLOCKED"
			}
			if envelope.Cell != cell || envelope.Context != peer.scope || envelope.Op != op || envelope.Protocol != guestbridge.TerminalProtocol() || string(envelope.Verification) != "null" || envelope.Payload.Status != status {
				t.Fatalf("invalid actual control reply: %s", response)
			}
		}
		return original, response, receipt
	}
	old := []terminalOldCell{}
	for i := 0; i < 80; i++ {
		op, value := "HELLO", any(guestbridge.TerminalProtocol())
		control := i >= 64
		if control {
			op = "RESUME"
			value = map[string]any{"previous_session": r.scope.Session, "ledger_hash": hashBytes([]byte("ledger")), "claims_hash": hashBytes([]byte("claims")), "domains_hash": hashBytes([]byte("domains"))}
		}
		r.cells++
		_, response, _ := exchange(0, r.cells, op, value, nil)
		nativeCell := uint64(0)
		for cell, command := range c.commands {
			if command.Context == r.scope && command.Delegation.GuestCell == r.cells {
				if nativeCell != 0 {
					t.Fatal("duplicate native original")
				}
				nativeCell = cell
			}
		}
		if nativeCell == 0 || len(c.responses[nativeCell]) == 0 {
			t.Fatal("missing actual native response")
		}
		old = append(old, terminalOldCell{guest: guestbridge.Ack{Cell: r.cells, ResponseHash: hashBytes(response)}, native: guestbridge.Ack{Cell: nativeCell, ResponseHash: hashBytes(c.responses[nativeCell])}, control: control})
	}
	type stats struct {
		Rows, Ordinary, Control, Old, Phase int
		High                                uint64
		Removed                             bool
		PackageBytes                        uint64 `json:"package_bytes"`
	}
	var before stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 0, "", 0), &before))
	if before.Rows != 80 || before.Ordinary != 64 || before.Control != 16 || before.High != 80 || before.PackageBytes == 0 {
		t.Fatalf("wrong actual exporter pre-cut: %+v", before)
	}
	request, donor, _, _ := terminalPublishCut(t, r, old, 0)
	r.cells++
	original, response, receipt := exchange(0, r.cells, "SESSION_RETIRE", request, donor)
	if receipt.Kind != "SESSION_RETIRED" || receipt.PollCount != 1 {
		t.Fatal("missing real signed terminal receipt")
	}
	var final stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 0, "", 0), &final))
	if final.Rows != 1 || final.Ordinary != 1 || final.Control != 0 || final.Old != 80 || !final.Removed || final.Phase != 3 || final.High != 81 || final.PackageBytes != before.PackageBytes {
		t.Fatalf("wrong actual exporter final owner: %+v", final)
	}
	// Exact same-live replay retains one native cell and identical full bytes.
	_, replay, _ := exchange(0, r.cells, "SESSION_RETIRE", request, donor)
	if !bytes.Equal(response, replay) {
		t.Fatal("exporter replay changed signed bytes")
	}
	_, _, _ = exchange(1, 1, "HELLO", guestbridge.TerminalProtocol(), nil)
	var peer stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 1, "", 0), &peer))
	if peer.High != 82 || peer.Rows != 1 || peer.Ordinary != 2 || peer.Control != 0 {
		t.Fatalf("shared native allocator stopped or reset: %+v", peer)
	}
	c.close()
	state, cells := terminalInspect(t, r)
	if state.Ordinary != 2 || state.Control != 0 || state.Workspace || len(cells) != 2 || state.Terminal[r.scope.Session] != 81 {
		t.Fatalf("wrong actual C8 final cut: %+v", state)
	}
	found := false
	for _, cell := range cells {
		if cell.Header.Cell != 81 {
			continue
		}
		found = true
		if cell.Terminal == nil || cell.Terminal.Phase != "TERMINAL_RESPONDED" || cell.Terminal.OldOrdinary != 64 || cell.Terminal.OldControl != 16 || len(cell.Terminal.Donors) != 79 {
			t.Fatal("C8 lost full original manifest/class owners")
		}
	}
	if !found {
		t.Fatal("final native C8 owner missing")
	}
	_, err := guestbridge.VerifyMappedReceipt(r.mappedBinding(), original, response)
	require(t, err)
}

		t.Fatal("actual guest80 handoff lost retained native manifest/classes")
	}
}

// Exporter-only boundary, as in Backend80: kernel/observer are substituted.
// Every native request, durable C8 final, signature and exporter owner is real.
func TestControlledReceivedActualTerminalPermanentExhaustion(t *testing.T) {
 r := newSourceRigDomainsUnstarted(t, guestbridge.TerminalProtocol(), 17, receivedInitialIDMap)
 c := startReceivedCarrierMode(t, r, true)
 responses := make([][]byte, 16)
 originals := make([][]byte, 16)
 type stats struct { Rows, Ordinary, Control, Old, Phase int; High uint64; Removed bool; PackageBytes uint64 `json:"package_bytes"` }
 for i, peer := range r.peers[:16] {
  request, donor, _, _ := terminalPublishCut(t, peer, nil, -1)
  original := mustRuntimeJSON(t, map[string]any{"cell": uint64(1), "context": peer.scope, "delegation": nil, "op": "SESSION_RETIRE", "payload": request, "protocol": guestbridge.TerminalProtocol(), "reclaim": donor})
  response := c.action("BACKEND_EXCHANGE", i, hex.EncodeToString(original), 0)
  receipt, err := guestbridge.VerifyMappedReceipt(peer.mappedBinding(), original, response); require(t, err)
  if receipt.Kind != "SESSION_RETIRED" { t.Fatal("missing actual terminal proof") }
  originals[i], responses[i] = original, response
 }
 beforeCommands := len(c.commands)
 // All 16 session entries are permanent. Actual admission must fail without
 // forwarding anything or replacing an old package/identity.
 c.action("BACKEND_ADD", 0, hex.EncodeToString(mustRuntimeJSON(t, r.peers[16].scope)), -28)
 c.action("BACKEND_ADD", 0, hex.EncodeToString(mustRuntimeJSON(t, r.peers[16].scope)), -28)
 for i := range r.peers[:16] {
  response := c.action("BACKEND_EXCHANGE", i, hex.EncodeToString(originals[i]), 0)
  if !bytes.Equal(response, responses[i]) { t.Fatal("permanent signed bytes changed") }
  var s stats; require(t, json.Unmarshal(c.action("BACKEND_STATS", i, "", 0), &s))
  if s.Rows != 1 || s.Ordinary != 0 || s.Control != 16 || s.High != 16 || !s.Removed || s.Phase != 3 || s.PackageBytes != 88504 { t.Fatalf("permanent exporter owners evicted/refunded: %+v", s) }
 }
 if len(c.commands) != beforeCommands || beforeCommands != 16 { t.Fatal("exhaustion/replay issued another native cell") }
 c.close()
 state, cells := terminalInspect(t, r)
 if state.Control != 16 || state.Ordinary != 0 || state.Workspace || len(cells) != 16 || len(state.Terminal) != 16 { t.Fatalf("permanent C8 owners changed: %+v", state) }
 for _, cell := range cells { if cell.Terminal == nil || cell.Terminal.Phase != "TERMINAL_RESPONDED" || !cell.Control { t.Fatal("lost permanent native final") } }
}

// Exercise actual hfs_call and unchanged VFS/workqueue cleanup wrappers against
// the real exporter/host. Exhaustion must not consume a claim or native cell.
func TestControlledReceivedActualCleanupByteFunding(t *testing.T) {
	for _, protocol := range []guestbridge.Protocol{guestbridge.MappedProtocol(), guestbridge.TerminalProtocol()} {
		t.Run(fmt.Sprint(protocol.Features), func(t *testing.T) {
			r := newSourceRigDomainsUnstarted(t, protocol, 2, receivedInitialIDMap)
			c := startReceivedCarrier(t, r)
			setup := func(session int, wrappers bool) (guestbridge.Capability, guestbridge.Capability) {
				request := func(op string, args any) []byte {
					return c.action("REQUEST", session, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
				}
				root := receivedResult(t, request("ROOT", map[string]any{}))
				c.action("ADOPT_ROOT_PREPARE", session, "", 0)
				caps := receivedResult(t, request("CREATE", map[string]any{"parent": root[0], "name": fmt.Sprintf("636c65616e7570%02x", session+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
				var description, path guestbridge.Capability
				for _, cap := range caps {
					if cap.Kind == "DESCRIPTION" {
						description = cap
					}
					if cap.Kind == "PATH" {
						path = cap
					}
				}
				if description.Kind == "" || path.Kind == "" {
					t.Fatal("missing actual CREATE capability")
				}
				if wrappers {
					c.action("CLEANUP_OWNERS", session, "", 0)
				} else {
					c.action("ADOPT_DESCRIPTION", session, "CREATE", 0)
				}
				c.action("REPLIES_DROP", session, "", 0)
				c.action("RETIRE", session, "", 0)
				return description, path
			}
			description, path := setup(0, false)
			operations := []struct {
				op   string
				args any
			}{
				{"DESCRIPTION_CLOSE", map[string]any{"description": description, "process": r.peers[0].credentials.Process, "close_sequence": 1}},
				{"DESCRIPTION_RELEASE", map[string]any{"description": description, "release_sequence": 1}},
				{"FORGET", map[string]any{"object": path, "count": 1}},
				{"PROCESS_EXIT", map[string]any{"process": r.peers[0].credentials.Process, "exit_sequence": 1}},
			}
			c.action("BYTE_PRESSURE", 0, "ordinary", 0)
			before := len(c.commands)
			for _, item := range operations {
				c.action("CLEANUP_REQUEST", 0, item.op+" "+hex.EncodeToString(mustRuntimeJSON(t, item.args)), -28)
			}
			if len(c.commands) != before {
				t.Fatal("unfunded cleanup emitted native bytes")
			}
			c.action("BYTE_UNPRESSURE", 0, "ordinary", 0)
			c.action("BYTE_PRESSURE", 0, "control", 0)
			for _, item := range operations {
				c.action("CLEANUP_REQUEST", 0, item.op+" "+hex.EncodeToString(mustRuntimeJSON(t, item.args)), 0)
				// The full reply remains live through the C owner/retry/retirement checks.
				// Reap it before the next full request, within the unchanged byte budget.
				c.action("REPLIES_DROP", 0, "", 0)
				c.action("RETIRE", 0, "", 0)
			}
			c.action("BYTE_UNPRESSURE", 0, "control", 0)
			setup(1, true)
			c.action("BYTE_PRESSURE", 0, "ordinary", 0)
			before = len(c.commands)
			c.action("CALLERS_UNFUNDED", 1, "", 0)
			if len(c.commands) != before {
				t.Fatal("unfunded production caller emitted native bytes")
			}
			c.action("BYTE_UNPRESSURE", 0, "ordinary", 0)
			c.action("BYTE_PRESSURE", 0, "control", 0)
			for _, item := range operations {
				c.action("CLEANUP_CALLER", 1, item.op, 0)
				c.action("RETIRE", 1, "", 0)
			}
			c.action("BYTE_UNPRESSURE", 0, "control", 0)
			seen := map[string]int{}
			for _, command := range c.commands {
				if command.Context != r.scope || command.Op != "CALL" {
					continue
				}
				var call guestbridge.Call
				require(t, json.Unmarshal(command.Payload, &call))
				for _, item := range operations {
					if call.Intent.Opcode == item.op {
						seen[item.op]++
						if command.Delegation.GuestOp != "CALL" || !bytes.Equal(call.Intent.Args, mustRuntimeJSON(t, item.args)) || !bytes.Equal(mustRuntimeJSON(t, call.Credentials), mustRuntimeJSON(t, r.peers[0].credentials)) {
							t.Fatalf("cleanup changed original wire/credentials: %s", command.Payload)
						}
					}
				}
			}
			for _, item := range operations {
				if seen[item.op] != 1 {
					t.Fatalf("cleanup %s executed %d times", item.op, seen[item.op])
				}
			}
			c.close()
		})
	}
}
