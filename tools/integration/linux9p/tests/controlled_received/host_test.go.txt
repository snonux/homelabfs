type receivedCarrier struct {

	rig                 *sourceRig

func startReceivedCarrier(t *testing.T, r *sourceRig) *receivedCarrier {
	return startReceivedCarrierMode(t, r, false)
}
func startReceivedCarrierMode(t *testing.T, r *sourceRig, backendOnly bool) *receivedCarrier {
	return startReceivedCarrierBudget(t, r, backendOnly, GuestArtifactBudget{OrdinaryBytes: 256 << 20, ControlBytes: 256 << 20, Reservations: 8192, Cookies: 4096, Received: 4096})
}
func startReceivedCarrierBudget(t *testing.T, r *sourceRig, backendOnly bool, budget GuestArtifactBudget) *receivedCarrier {
	t.Helper()
	// Both paths are explicit: a materialized test must fail on absent inputs.
	build, source := os.Getenv("HFS_C6_BUILD"), os.Getenv("HFS_C6_SOURCE")
	if build == "" || source == "" {
		t.Fatal("HFS_C6_BUILD and HFS_C6_SOURCE required for actual C6 consumer")
	}
	group := GuestProvisionGroup{Authorization: r.artifactAuthorization}
	credentials := make([]guestbridge.Credentials, len(r.peers))
	contexts := []guestbridge.Context{}
	for i, peer := range r.peers {
		credentials[i] = peer.credentials
		if i < 16 { contexts = append(contexts, peer.scope) }
		group.Signers = append(group.Signers, GuestArtifactSigners{Host: peer.key, Guest: r.artifactGuest})
		group.LocalMounts = append(group.LocalMounts, GuestArtifactMount{DomainIndex: i, Session: peer.scope.Session, DeviceTag: fmt.Sprintf("hfs%d", i), DeviceName: fmt.Sprintf("virtio%d", i), UserNamespace: peer.credentials.UserNamespace, IDMapHash: peer.credentials.IDMapHash, Realm: testUUID(100).String(), Subject: "616263", Anchor: testUUID(101).String(), Nonce: testUUID(103).String(), ViewControlNonce: r.artifactAuthorization.Enrollment.TicketFamilies[i].Control.Nonce})
	}
	provision, err := SignGuestProvision(r.scope.GuestBoot, r.installed.Admission.ProviderNonce, []GuestProvisionGroup{group}, budget, r.artifactOperator)
	require(t, err)
	bind, err := SignGuestBind(provision, group.Authorization.Enrollment.ID.String(), r.supervisor.record.Controller.String(), r.artifactOperator, group.Signers)
	require(t, err)
	output := filepath.Join(r.temporary(), "build")
	compiler := exec.CommandContext(t.Context(), "python3", build, "--source", source, "--output", output, "--operator-key", hex.EncodeToString(r.artifactOperator.Public().(ed25519.PublicKey)))
	log, err := compiler.CombinedOutput()
	if err != nil {
		t.Fatalf("actual C6 build: %v\n%s", err, log)
	}
	consumer := filepath.Join(output, "consumer")
	if _, err := os.Stat(consumer); err != nil {
		t.Fatalf("missing actual C6 consumer: %v", err)
	}
	c := &receivedCarrier{t: t, rig: r, backendOnly: backendOnly, commands: map[uint64]guestbridge.Command{}, originals: map[string][]byte{}, responses: map[uint64][]byte{}}
	c.cmd = exec.CommandContext(t.Context(), consumer)
	ledger := r.temporary()
	require(t, os.Chmod(ledger, 0700))
	c.cmd.Env = append(os.Environ(), "HFS_C6_LEDGER="+ledger, "ASAN_OPTIONS=detect_leaks=0:allocator_may_return_null=1:quarantine_size_mb=16")
	if backendOnly {
		c.cmd.Env = append(c.cmd.Env, "HFS_C6_BACKEND_ONLY=1")
	}
	c.input, err = c.cmd.StdinPipe()
	require(t, err)
	stdout, err := c.cmd.StdoutPipe()
	require(t, err)
	c.reader = bufio.NewScanner(stdout)
	c.reader.Buffer(make([]byte, 65536), 4*guestbridge.BodyBytes+256)
	c.cmd.Stderr = &c.diagnostics
	require(t, c.cmd.Start())
	t.Cleanup(func() {
		if c.cmd.ProcessState == nil {
			_ = c.cmd.Process.Kill()
			_ = c.cmd.Wait()
		}
	})
	c.write(string(mustRuntimeJSON(t, map[string]any{"base": *r.mappingNativeCell, "contexts": contexts, "protocol_features": guestbridge.TerminalProtocol().Features, "binds": []json.RawMessage{bind}, "credentials": credentials, "os": map[string]any{"controller": r.supervisor.record.Controller.String(), "native_endpoint": r.native.Endpoint, "boot_id": r.scope.GuestBoot, "provider_nonce": r.installed.Admission.ProviderNonce, "capture_tuples": r.credentialCaptures}, "provision": json.RawMessage(provision)})))
	c.readUntil(0, true)
	return c
}
func (d *receivedDiagnostics) Write(raw []byte) (int, error) {
	d.mu.Lock()
	defer d.mu.Unlock()
	return d.buffer.Write(raw)
}
func (d *receivedDiagnostics) String() string {
	d.mu.Lock()
	defer d.mu.Unlock()
	return d.buffer.String()
}

func (c *receivedCarrier) write(line string) {
	c.t.Helper()
	_, err := fmt.Fprintln(c.input, line)
	require(c.t, err)
}
func (c *receivedCarrier) number(s string) uint64 {
	c.t.Helper()
	n, err := strconv.ParseUint(s, 10, 64)
	require(c.t, err)
	return n
}
func (c *receivedCarrier) readUntil(id uint64, startup bool) (int, []byte) {
	c.t.Helper()
	for c.reader.Scan() {
		fields := strings.Fields(c.reader.Text())
		if len(fields) == 1 && fields[0] == "READY" && startup {
			return 0, nil
		}
		if len(fields) < 2 {
			c.t.Fatalf("invalid C6 IPC %q", c.reader.Text())
		}
		switch fields[0] {
		case "L":
			c.t.Log(strings.Join(fields[1:], " "))
		case "S":
			if c.backendOnly {
				continue
			} // This mode explicitly substitutes the OS observer.
			if len(fields) != 2 {
				c.t.Fatal("snapshot arity")
			}
			raw, err := hex.DecodeString(fields[1])
			require(c.t, err)
			var admission kernelAdmission
			require(c.t, strictJSON(raw, &admission))
			// Replace only the complete actual serializer record; static observations
			// retain their independently measured manifest, notes, boot and ABI fields.
			c.rig.installed.Admission = admission
			next := c.rig.observationFile + ".next"
			require(c.t, os.WriteFile(next, mustRuntimeJSON(c.t, c.rig.installed), 0600))
			require(c.t, os.Rename(next, c.rig.observationFile))
		case "W":
			c.nativeWrite(fields)
		case "R":
			c.nativeRead(fields)
		case "Y":
			yielded := c.number(fields[1])
			if yielded != id && !startup {
				c.t.Fatalf("unexpected scheduling ID %d want %d", yielded, id)
			}
			if c.yields != nil {
				c.yields(yielded)
			}
			c.write("CONTINUE")
		case "D":
			if startup || len(fields) != 4 || c.number(fields[1]) != id {
				c.t.Fatalf("unexpected completion %q", c.reader.Text())
			}
			code, err := strconv.Atoi(fields[2])
			require(c.t, err)
			if fields[3] == "-" {
				return code, nil
			}
			raw, err := hex.DecodeString(fields[3])
			require(c.t, err)
			return code, raw
		default:
			c.t.Fatalf("unexpected IPC %q", c.reader.Text())
		}
	}
	c.t.Fatalf("C6 ended before completion: %v %s", c.reader.Err(), c.diagnostics.String())
	return -1, nil
}
func (c *receivedCarrier) nativeWrite(fields []string) {
	c.t.Helper()
	if len(fields) != 3 {
		c.t.Fatal("native write arity")
	}
	offset := c.number(fields[1])
	raw, err := hex.DecodeString(fields[2])
	require(c.t, err)
	frame, err := guestbridge.DecodeFrame(raw)
	require(c.t, err)
	if frame.Offset != 0 || uint32(len(frame.Data)) != frame.Total {
		c.t.Fatal("unexpected fragmented C6 command")
	}
	command, err := guestbridge.DecodeCommandVersion(frame.Data, 4)
	require(c.t, err)
	if command.Delegation == nil {
		c.t.Fatal("C6 native request omitted actual delegation")
	}
	d := command.Delegation
	original := command
	original.Cell = d.GuestCell
	original.Op = d.GuestOp
	original.Reclaim = d.GuestReclaim
	original.Delegation = nil
	if original.Op == "RELEASE_CELL" {
		original.Payload = mustRuntimeJSON(c.t, d.GuestReclaim)
	}
	key := fmt.Sprintf("%s/%d", command.Context.Session, d.GuestCell)
	originalBytes := mustRuntimeJSON(c.t, original)
	if previous := c.originals[key]; previous != nil && !bytes.Equal(previous, originalBytes) {
		c.t.Fatal("C6 continuation changed full original credentials/payload/Ack")
	}
	c.originals[key] = originalBytes
	if old, ok := c.commands[frame.Cell]; ok && !bytes.Equal(mustRuntimeJSON(c.t, old), frame.Data) {
		c.t.Fatal("actual native retry changed immutable request bytes")
	}
	c.commands[frame.Cell] = command
	n, err := c.rig.mappingTransport.Write(c.t.Context(), c.rig.native, c.rig.native.Endpoint.CommandNode, offset, raw)
	if err != nil {
		c.write("ERROR")
	} else {
		c.write(fmt.Sprintf("OK %d", n))
	}
}
func (c *receivedCarrier) nativeRead(fields []string) {
	c.t.Helper()
	if len(fields) != 4 {
		c.t.Fatal("native read arity")
	}
	offset, count := c.number(fields[2]), c.number(fields[3])
	if c.holdNativeResponses {
		// Declared OS delivery delay only; no response or receipt is fabricated.
		c.nativeAgain++
		c.write("AGAIN")
		return
	}
	node := c.rig.native.Endpoint.ResultNode
	if fields[1] == "12" {
		node = c.rig.native.Endpoint.RecallNode
	} else if fields[1] != "11" {
		c.t.Fatal("unknown native fd")
	}
	raw, err := c.rig.mappingTransport.Read(c.t.Context(), c.rig.native, node, offset, uint32(count))
	if errors.Is(err, fsapi.WouldBlock) {
		c.nativeAgain++
		c.write("AGAIN")
		return
	}
	if c.passReadErrors && errors.Is(err, fsapi.IO) {
		// Since z54 a cell the host will never answer (blocked) reads as
		// a definitive EIO instead of WouldBlock until the guest's
		// deadline; pass it on like FUSE does (bh4). Only tests that
		// expect such a cell opt in; anywhere else IO stays fatal.
		c.nativeErrors++
		c.write("ERROR")
		return
	}
	require(c.t, err)
	if offset%guestbridge.CellStride == guestbridge.HeaderBytes {
		c.responses[offset/guestbridge.CellStride] = bytes.Clone(raw)
	}
	c.write("DATA " + hex.EncodeToString(raw))
}
func (c *receivedCarrier) action(verb string, session int, tail string, want int) []byte {
	c.t.Helper()
	c.nextID++
	id := c.nextID
	c.write(strings.TrimSpace(fmt.Sprintf("%s %d %d %s", verb, id, session, tail)))
	code, raw := c.readUntil(id, false)
	if code != want {
		for cell, command := range c.commands {
			if command.Op == "GAP_RESOLVE" || command.Op == "QUERY" || command.Op == "RETIRE" {
				c.t.Logf("actual retained exchange cell%d %s request=%s response=%s", cell, command.Op, command.Payload, c.responses[cell])
			}
		}
		c.t.Fatalf("C6 %s session%d %s = %d want%d body=%s diagnostics=%s", verb, session, tail, code, want, raw, c.diagnostics.String())
	}
	return raw
}
func (c *receivedCarrier) run(session int, reserved bool, op string, payload any) []byte {
	c.t.Helper()
	lane := 0
	if reserved {
		lane = 1
	}
	return c.action("RUN", session, fmt.Sprintf("%d %s %s", lane, op, hex.EncodeToString(mustRuntimeJSON(c.t, payload))), 0)
}
func (c *receivedCarrier) call(session int, op string, args any) []byte {
	c.t.Helper()
	peer := c.rig.peers[session]
	peer.claims++
	intent := guestbridge.Intent{Opcode: op, Args: mustRuntimeJSON(c.t, args)}
	call := guestbridge.Call{Claim: guestbridge.Claim{Realm: testUUID(100).String(), Subject: "616263", Anchor: testUUID(101).String(), Nonce: testUUID(103).String(), Number: peer.claims, OriginalHash: hashBytes(mustRuntimeJSON(c.t, intent))}, Credentials: peer.credentials, Intent: intent}
	c.run(session, false, "ALLOCATE", call)
	c.run(session, false, "CALL", call)
	return c.run(session, true, "QUERY", map[string]any{"claim": call.Claim})
}
func (c *receivedCarrier) close() {
	c.t.Helper()
	c.write("END")
	require(c.t, c.input.Close())
	require(c.t, c.cmd.Wait())
}

// TestControlledReceivedActualProviderProducer drives the actual provider
// checks, a multipoll VIEW_WAIT EVENT receipt and a recall of each domain.
//
// Since task 784 a clean DOMAIN READ release recalls only the releasing
// domain, so the release nested in session 0's WAIT is session 0's own
// lease (it used to be session 1's, whose all-domain recall also reached
// session 0). Session 1's lease is released afterwards and only session 1
// takes part in that recall (bh4).
func TestControlledReceivedActualProviderProducer(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	domains := receivedProviderSetup(t, c)
	receivedMultipollEvent(t, c, domains[0])
	c.action("CHECK", 0, "event", 0)
	c.action("CHECK", 0, "forged_event", 0)
	c.action("CHECK", 0, "wrong_channel", 0)
	// The WAIT above already received session 0's STOPPED event.
	receivedRecallPhases(c, 0, domains[0], false)
	c.action("RELEASE_START", 1, "", 0)
	receivedRecallPhases(c, 1, domains[1], true)
	for i := range 2 {
		c.action("CHECK", i, "stale_target", 0)
		c.action("CHECK", i, "cap", 0)
		c.action("CHECK", i, "cookie_balance", 0)
	}
	c.close()
}

// receivedProviderSetup adopts each session's ROOT/DOMAIN and a created
// file's DESCRIPTION, acquires a DOMAIN READ lease and runs the provider's
// stored-object checks. It returns the DOMAIN capabilities.
func receivedProviderSetup(t *testing.T, c *receivedCarrier) []guestbridge.Capability {
	t.Helper()
	for i := range 2 {
		c.action("RELEASE_NEGATIVES", i, "", 0)
	}
	domains := make([]guestbridge.Capability, 2)
	for i := range 2 {
		root := c.call(i, "ROOT", map[string]any{})
		caps := receivedResult(t, root)
		require(t, json.Unmarshal(c.action("ADOPT_ROOT", i, "", 0), &domains[i]))
		created := c.call(i, "CREATE", map[string]any{"parent": caps[0], "name": fmt.Sprintf("66696c65%02x", i+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}})
		receivedResult(t, created)
		c.action("ADOPT_DESCRIPTION", i, "CREATE", 0)
	}
	for i := range 2 {
		c.action("ACQUIRE", i, "", 0)
		for _, check := range []string{"target", "publication", "cap", "cookie_balance"} {
			c.action("CHECK", i, check, 0)
		}
		for _, check := range []string{"forged_target", "forged_event", "forged_publication", "nonnull_publication", "wrong_channel", "bad_device", "bad_map", "retirement_refuses"} {
			c.action("CHECK", i, check, 0)
		}
	}
	return domains
}

// receivedRecallPhases acknowledges every phase of a recall of one session's
// domain, each after the actual VIEW_WAIT that receives its event; with
// waitFirst false the STOPPED event was already received.
func receivedRecallPhases(c *receivedCarrier, session int, domain guestbridge.Capability, waitFirst bool) {
	c.t.Helper()
	for _, phase := range []string{"STOPPED", "INSTALLED_DRAIN", "DRAINED", "INSTALLED_FINAL", "REOPENED"} {
		if waitFirst || phase != "STOPPED" {
			c.run(session, true, "VIEW_WAIT", map[string]any{"domain": domain, "after_sequence": 0})
		}
		c.action("PHASE", session, phase, 0)
	}
}

// receivedMultipollAttempts bounds receivedMultipollEvent's retries.
const receivedMultipollAttempts = 4

// receivedMultipollEvent proves that session 0's WAIT retains the exact
// prior EMPTY Ack across actual internal polls until the recall EVENT of a
// release nested in that WAIT arrives. lib9p bounds one WAIT to 1 s
// (hfs9p.c _wait_step); under -race, ASan and a loaded host the nested
// release alone has taken 0.3-0.7 s, so the event can miss that window and
// the WAIT legitimately ends EMPTY (bh4). Such an attempt finishes its
// recall, releases the lease, retires the view and takes a fresh lease, then
// tries again. Only the multipoll EVENT itself is verified, never an EMPTY.
//
// VIEW_RETIRE is required: a released lease's tickets still hold mount
// capacity until the view retirement retires them (observed behaviour; the
// exact charge was not isolated). A mounted guest retires them in the
// background; this harness has no background controller. Without the
// retirement the fourth ACQUIRE of a session failed with -ENOSPC (bh4
// review 1). With it, and the bound temporarily raised to eight for the
// check, six forced misses in a row (a 1.2 s sleep before the nested
// release) still ended in a verified EVENT on the seventh attempt. The
// committed bound is four attempts, so up to three misses are tolerated; it only caps how long a host too loaded
// for the 1 s window is retried.
func receivedMultipollEvent(t *testing.T, c *receivedCarrier, domain guestbridge.Capability) {
	t.Helper()
	for attempt := 1; ; attempt++ {
		empty := c.run(0, true, "VIEW_WAIT", map[string]any{"domain": domain, "after_sequence": 0})
		if receivedWaitStatus(t, empty) != "EMPTY" {
			t.Fatalf("initial actual WAIT must expire EMPTY: %s", empty)
		}
		event := receivedNestedReleaseWait(t, c, domain)
		if receivedWaitStatus(t, event) == "EVENT" {
			verifyReceivedWait(t, c, 0, empty, event)
			return
		}
		if attempt == receivedMultipollAttempts {
			t.Fatalf("actual multipoll EVENT receipt missing after %d attempts: %s", attempt, event)
		}
		t.Logf("attempt %d: the nested release's EVENT missed the 1 s WAIT window; finishing that recall and retrying", attempt)
		receivedRecallPhases(c, 0, domain, true)
		c.action("RELEASE_DONE", 0, "", 0)
		c.action("VIEW_RETIRE", 0, "", 0)
		c.action("ACQUIRE", 0, "", 0)
	}
}

// receivedNestedReleaseWait runs session 0's WAIT and, at its first
// scheduling yield after a continuation poll was sent (the reclaim poll was
// answered EMPTY), releases session 0's lease through the actual control
// path. It returns the WAIT's response.
func receivedNestedReleaseWait(t *testing.T, c *receivedCarrier, domain guestbridge.Capability) []byte {
	t.Helper()
	before := c.nativeAgain
	waitBase := uint64(0)
	for cell := range c.commands {
		waitBase = max(waitBase, cell)
	}
	nested := false
	c.yields = func(_ uint64) {
		for cell, command := range c.commands {
			if cell > waitBase && command.Context == c.rig.scope && command.Delegation.GuestOp == "VIEW_WAIT" && command.Delegation.PollIndex >= 1 {
				nested = true
				break
			}
		}
		if nested {
			c.yields = nil
			c.action("RELEASE_START", 0, "", 0)
		}
	}
	event := c.run(0, true, "VIEW_WAIT", map[string]any{"domain": domain, "after_sequence": 0})
	c.yields = nil
	if !nested || c.nativeAgain <= before {
		t.Fatal("WAIT did not yield after actual backend EAGAIN")
	}
	return event
}

func receivedWaitStatus(t *testing.T, response []byte) string {
	t.Helper()
	var reply struct{ Payload struct{ Status string } }
	require(t, json.Unmarshal(response, &reply))
	return reply.Payload.Status
}
func verifyReceivedWait(t *testing.T, c *receivedCarrier, session int, previous, response []byte) {
	t.Helper()
	var reply struct {
		Cell         uint64
		Verification *guestbridge.VerificationReceiptV2
	}
	require(t, json.Unmarshal(response, &reply))
	if reply.Verification == nil || reply.Verification.Kind != "EVENT" || reply.Verification.PollCount < 2 {
		t.Fatalf("actual multipoll EVENT receipt missing: %s", response)
	}
	key := fmt.Sprintf("%s/%d", c.rig.peers[session].scope.Session, reply.Cell)
	original := c.originals[key]
	_, err := guestbridge.VerifyMappedReceipt(c.rig.peers[session].mappedBinding(), original, response)
	require(t, err)
	var command guestbridge.Command
	require(t, json.Unmarshal(original, &command))
	var prior struct{ Cell uint64 }
	require(t, json.Unmarshal(previous, &prior))
	if command.Reclaim == nil || command.Reclaim.Cell != prior.Cell || command.Reclaim.ResponseHash != hashBytes(previous) {
		t.Fatal("WAIT did not retain exact prior EMPTY response Ack")
	}
	legs := 0
	for _, native := range c.commands {
		d := native.Delegation
		if native.Context.Session != command.Context.Session || d.GuestCell != command.Cell {
			continue
		}
		legs++
		if d.GuestReclaim == nil || *d.GuestReclaim != *command.Reclaim {
			t.Fatal("native poll changed original Ack")
		}
	}
	if legs != int(reply.Verification.PollCount) {
		t.Fatalf("real poll count=%d signed=%d", legs, reply.Verification.PollCount)
	}
	var native, guest map[string]json.RawMessage
	require(t, json.Unmarshal(c.responses[reply.Verification.Cell], &native))
	require(t, json.Unmarshal(response, &guest))
	if !bytes.Equal(native["verification"], guest["verification"]) {
		t.Fatal("C carrier changed Producer signature bytes")
	}
}

// TestControlledReceivedActualViewAdvance (xf4, task 4h4): the guest side
// of a guest-requested view advance with the real 0005 provider. It needs
// the xf4 host (task 3h4): an older host refuses "advance" in the request.
// After the initial lease's release (its domain recall acknowledged phase by
// phase) a CREATE moves the host cut; a cacheless
// observation's VIEW_ACQUIRE with advance is granted on the current cut and
// adopted (ADVANCE). Its VIEW_QUERY replay is current for the advanced view
// and accepted as an ordinary lease (ADVANCE_REPLAY). After its release
// (no recall: an Advance lease) a further CREATE and a second advance move

func TestControlledReceivedActualSourceCorruption(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	c.call(0, "ROOT", map[string]any{})
	c.action("ADOPT_ROOT", 0, "", 0)
	// Establish an actual retained source before changing its actual journal bytes.
	c.action("ACQUIRE", 0, "", 0)
	before := uint64(0)
	for cell := range c.commands {
		if cell > before {
			before = cell
		}
	}
	corruptReceivedJournal(t, r)
	// The real Producer withholds the damaged SOURCE: the actual native worker
	// publishes no response. Since z54 that blocked cell reads as a definitive
	// EIO (lib9p: Rbridge ERROR INVALID), so the actual exchange fails at once
	// with -EPROTO instead of reaching its -ETIMEDOUT deadline (bh4).
	errorsBefore := c.nativeErrors
	c.passReadErrors = true
	c.action("ACQUIRE", 0, "", -71)
	c.passReadErrors = false
	if c.nativeErrors == errorsBefore {
		t.Fatal("damaged source was not refused with a definitive native read error")
	}
	attempts := 0
	for cell, command := range c.commands {
		if cell <= before {
			continue
		}
		if command.Op == "VIEW_ACQUIRE" {
			attempts++
			t.Logf("corrupted-source actual VIEW_ACQUIRE native cell %d: no response=%t", cell, c.responses[cell] == nil)
			if c.responses[cell] != nil {
				t.Fatal("damaged source minted an actual response")
			}
		}
	}
	if attempts != 1 {
		t.Fatalf("corruption must reach exactly one real new source RPC: %d", attempts)
	}
	c.action("CHECK", 0, "cookie_balance", 0)
	c.close()
}
func corruptReceivedJournal(t *testing.T, r *sourceRig) {
	t.Helper()
	stream := r.artifactAuthorization.Enrollment.Domains[0].Journal.Stream
	recovered, err := segment.Recover(t.Context(), r.source, stream, segment.RecoveryOptions{})
	require(t, err)
	if len(recovered.Segments) == 0 {
		t.Fatal("no actual source journal segment")
	}
	f, err := r.source.Open(t.Context(), recovered.Segments[0].Name, os.O_RDWR, 0)
	require(t, err)
	defer func() { require(t, f.Close()) }()
	var original [1]byte
	n, err := f.ReadAt(t.Context(), original[:], 32)
	require(t, err)
	if n != 1 {
		t.Fatal("short actual journal read")
	}
	changed := original
	changed[0] ^= 1
	n, err = f.WriteAt(t.Context(), changed[:], 32)
	require(t, err)
	if n != 1 {
		t.Fatal("short actual journal corruption")
	}
	require(t, f.Sync(t.Context()))
	var actual [1]byte
	n, err = f.ReadAt(t.Context(), actual[:], 32)
	require(t, err)
	if n != 1 || actual != changed || actual == original {
		t.Fatal("actual journal bytes did not change")
	}
	t.Logf("actual source journal %s offset32 byte %02x -> %02x (synced and reread)", recovered.Segments[0].Name, original[0], actual[0])
}

// receivedDirtyRelease releases session 0's EXCLUSIVE_DIRTY lease through
// the complete production hfs_lease_release (RELEASE_DONE). That recall stays
// all-domain, so both sessions take part and session 0 freezes and drains its
// page. Session 1's clean PAGE READ release (RELEASE_START 1) used to start
// this recall; since task 784 such a recall is scoped to domain 1 and session
// 0 never gets an event (bh4). The abbreviated RELEASE_START cannot be used
// for session 0: it never admits its initial ticket, so the freeze ticket of
// the same family would wait for it forever. As in the terminal dirty
// scenario, the recall runs nested in the release's own scheduling yields once
// its ticket is admitted (RELEASE_READY), and CLOSE_PENDING follows that
// admission: a closing session sends only its exempt release/drain work.
func receivedDirtyRelease(t *testing.T, c *receivedCarrier, domains []guestbridge.Capability, retainedPressure bool) {
	t.Helper()
	recalled := false
	c.yields = func(_ uint64) {
		var ready bool
		require(t, json.Unmarshal(c.action("RELEASE_READY", 0, "", 0), &ready))
		if !ready {
			return
		}
		c.yields = nil
		c.action("CLOSE_PENDING", 0, "", 0)
		receivedDirtyRecall(t, c, domains, retainedPressure)
		recalled = true
	}
	c.action("RELEASE_DONE", 0, "", 0)
	c.yields = nil
	if !recalled {
		t.Fatal("dirty lease release never yielded after its ticket admission")
	}
}

// receivedDirtyRecall acknowledges every phase of the all-domain recall in
// both sessions: session 0's STOPPED first refuses while a page is pinned,
// then freezes the dirty page, which DRAIN writes before DRAINED. With
// retained pressure each session holds four EVENT owners and its actors'
// control quota first, and both release their floors at the end.
func receivedDirtyRecall(t *testing.T, c *receivedCarrier, domains []guestbridge.Capability, retainedPressure bool) {
	t.Helper()
	for i := range 2 {
		if retainedPressure {
			for event := 0; event < 4; event++ {
				c.run(i, true, "VIEW_WAIT", map[string]any{"domain": domains[i], "after_sequence": 0})
				c.action("EVENT_HOLD", i, "", 0)
			}
			c.action("EVENT_RESTORE", i, "", 0)
			c.action("ACTOR_HOLD", i, "", 0)
		} else {
			c.run(i, true, "VIEW_WAIT", map[string]any{"domain": domains[i], "after_sequence": 0})
		}
		if i == 0 {
			c.action("PIN", i, "1", 0)
			c.action("PHASE", i, "STOPPED", -16)
			c.action("PIN", i, "0", 0)
		}
		c.action("PHASE", i, "STOPPED", 0)
	}
	c.action("DRAIN", 0, "", 0)
	for _, phase := range []string{"INSTALLED_DRAIN", "DRAINED", "INSTALLED_FINAL", "REOPENED"} {
		for i := range 2 {
			c.run(i, true, "VIEW_WAIT", map[string]any{"domain": domains[i], "after_sequence": 0})
			c.action("PHASE", i, phase, 0)
		}
	}
	for i := 0; retainedPressure && i < 2; i++ {
		c.action("FLOOR_UNHOLD", i, "", 0)
	}
}

// Reconstruct only the documented scalar translation of the recorded actual
// native receipt. This is an inert cross-language check; the real guest callback
// has already consumed the actual privately received response before this runs.
func checkReceivedViewFloorBindings(t *testing.T, c *receivedCarrier, extended bool) {
	t.Helper()
	count := 0
	blocked := 0
	var blockedKey string
	floors := make(map[string]struct {
		Through uint64
		Hashes  []string `json:"result_hashes"`
	})
	for cell, command := range c.commands {
		if command.Op != "VIEW_RETIRE" {
			continue
		}
		var fields map[string]json.RawMessage
		require(t, json.Unmarshal(c.responses[cell], &fields))
		var cut struct {
			Anchor, Nonce string
			Credentials   guestbridge.Credentials
			Through       uint64
			Hashes        []string `json:"result_hashes"`
		}
		require(t, json.Unmarshal(command.Payload, &cut))
		key := command.Context.Session + "/" + cut.Anchor + "/" + cut.Nonce
		var response struct {
			Payload struct {
				Status string
				Floor  uint64
				Proof  *string `json:"proof_hash"`
			}
		}
		require(t, json.Unmarshal(c.responses[cell], &response))
		var receipt guestbridge.VerificationReceiptV2
		require(t, json.Unmarshal(fields["verification"], &receipt))
		if receipt.Kind != "VIEW_FLOOR" {
			if response.Payload.Status != "BLOCKED" || response.Payload.Floor != 0 || response.Payload.Proof != nil ||
				command.Context != c.rig.peers[1].scope || cut.Through != 1 || len(cut.Hashes) != 1 || !bytes.Equal(mustRuntimeJSON(t, cut.Credentials), mustRuntimeJSON(t, c.rig.credentials)) {
				t.Fatalf("unexpected actual blocked view cut: %s / %s", command.Payload, c.responses[cell])
			}
			blocked++
			blockedKey = key
			continue
		}
		floors[key] = struct {
			Through uint64
			Hashes  []string `json:"result_hashes"`
		}{cut.Through, cut.Hashes}
		count++
		fields["cell"] = mustRuntimeJSON(t, receipt.GuestCell)
		fields["op"] = mustRuntimeJSON(t, "VIEW_RETIRE")
		original := c.originals[fmt.Sprintf("%s/%d", command.Context.Session, receipt.GuestCell)]
		if len(original) == 0 {
			t.Fatal("missing actual original VIEW_RETIRE")
		}
		binding := c.rig.peers[0].mappedBinding()
		for _, peer := range c.rig.peers {
			if peer.scope == command.Context {
				binding = peer.mappedBinding()
			}
		}
		_, err := guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields))
		require(t, err)
		changed := receipt
		changed.MappingHash = hashBytes([]byte("wrong mapping"))
		fields["verification"] = mustRuntimeJSON(t, changed)
		if _, err = guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted changed mapping")
		}
		changed = receipt
		changed.PollCount = 2
		fields["verification"] = mustRuntimeJSON(t, changed)
		if _, err = guestbridge.VerifyMappedReceipt(binding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted wrong poll mapping")
		}
		fields["verification"] = mustRuntimeJSON(t, receipt)
		wrongBinding := binding
		wrongBinding.Controller = testUUID(120).String()
		if _, err = guestbridge.VerifyMappedReceipt(wrongBinding, original, mustRuntimeJSON(t, fields)); err == nil {
			t.Fatal("VIEW_FLOOR accepted other controller")
		}
	}
	if !extended {
		if count != 4 || blocked != 0 {
			t.Fatalf("pressure view floors/blocked %d/%d", count, blocked)
		}
		return
	}
	if count != 5 || blocked != 1 {
		t.Fatalf("actual signed view floors/blocked cuts %d/%d", count, blocked)
	}
	final := floors[blockedKey]
	if final.Through != 2 || len(final.Hashes) != 2 {
		t.Fatalf("final control floor omitted real releases: %+v", final)
	}
	actual := make(map[string]bool)
	for cell, command := range c.commands {
		if command.Op != "VIEW_RELEASE" || command.Context != c.rig.peers[1].scope {
			continue
		}
		var reply struct{ Payload struct{ Status, ID string } }
		require(t, json.Unmarshal(c.responses[cell], &reply))
		if reply.Payload.Status == "RELEASED" {
			actual[reply.Payload.ID] = true
		}
	}
	// Final release may be completed by VIEW_QUERY; its protected payload
	// retains the original VIEW_RELEASE operation and exact response self-id.
	for cell, command := range c.commands {
		if command.Op != "VIEW_QUERY" || command.Context != c.rig.peers[1].scope {
			continue
		}
		var reply struct {
			Payload struct{ Status, ID, Operation string }
		}
		require(t, json.Unmarshal(c.responses[cell], &reply))
		if reply.Payload.Status == "RELEASED" && reply.Payload.Operation == "VIEW_RELEASE" {
			actual[reply.Payload.ID] = true
		}
	}
	if len(actual) != 2 || !actual[final.Hashes[0]] || !actual[final.Hashes[1]] || final.Hashes[0] == final.Hashes[1] {
		t.Fatalf("control floor differs from actual release replies: %v / %v", final.Hashes, actual)
	}
}

// receivedAbandons returns the actual ABANDON commands of one session with
// their responses, in cell order.
func receivedAbandons(t *testing.T, c *receivedCarrier, session int) (payloads, responses [][]byte) {
	t.Helper()
	cells := []uint64{}
	for cell, command := range c.commands {
		if command.Op == "ABANDON" && command.Context.Session == c.rig.peers[session].scope.Session {
			cells = append(cells, cell)
		}
	}
	slices.Sort(cells)
	for _, cell := range cells {
		payloads = append(payloads, c.commands[cell].Payload)
		responses = append(responses, c.responses[cell])
	}
	return payloads, responses
}

// receivedAbandonOutcome decodes the single outcome of an ABANDON response.
func receivedAbandonOutcome(t *testing.T, raw []byte) (number uint64, status, proof string) {
	t.Helper()
	var reply struct {
		Payload struct {
			Claims []struct {
				Number uint64  `json:"number"`
				Status string  `json:"status"`
				Proof  *string `json:"proof_hash"`
			} `json:"claims"`
		}
		Verification json.RawMessage
	}
	require(t, json.Unmarshal(raw, &reply))
	if len(reply.Payload.Claims) != 1 || string(reply.Verification) != "null" {
		t.Fatalf("ABANDON reply shape: %s", raw)
	}
	outcome := reply.Payload.Claims[0]
	if outcome.Proof != nil {
		proof = *outcome.Proof
	}
	return outcome.Number, outcome.Status, proof
}

// hh4: a blocked session abandons its stranded never-prepared claim.
// Session 0: the CALL of claim 2 never reaches the host (FAULT CALL 1), so
// the host holds it ALLOCATED and charged. The block queues abandon_work; the
// worker skips the request while a caller still holds it (users) and is not
// queued again; the final pass (kill_sb's) then sends exactly one ABANDON with
// the exact claim, the host answers ABANDONED, the request becomes a gap and
// RETIRE passes it with the same proof GAP_RESOLVE returns. A further pass
// sends nothing. Session 1: the CALL of claim 2 reached the host and ran (its
// reply lost, FAULT CALL 2): ABANDON answers KEPT and the request stays
// retained. Rerunning it (REQUEST_CONTINUE restarts hfs_request_run at its
// CALL) sends nothing for it; on this blocked session the CALL is refused at
// admission anyway. The QUERY side of the promise (a request listed after its
// CALL was admitted) is tests/controlled mode abandon-query.
func TestControlledReceivedActualAbandon(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	args := "ROOT " + hex.EncodeToString(mustRuntimeJSON(t, map[string]any{}))
	intent := guestbridge.Intent{Opcode: "ROOT", Args: json.RawMessage(`{}`)}
	originalHash := hashBytes(mustRuntimeJSON(t, intent))
	for session := range 2 {
		c.action("REQUEST", session, args, 0)
		c.action("ADOPT_ROOT_PREPARE", session, "", 0)
		c.action("ABANDON_ARM", session, "", 0)
	}
	c.action("FAULT", 0, "CALL 1", 0)
	c.action("REQUEST", 0, args, -5)
	if floor, requests, unknown := receivedState(t, c, 0); floor != 0 || requests != 2 || unknown != 1 {
		t.Fatalf("stranded claim lost: %d/%d/%d", floor, requests, unknown)
	}
	c.action("ABANDON_USERS", 0, "1", 0)
	if ran := string(c.action("ABANDON_STEP", 0, "", 0)); ran != "true" {
		t.Fatalf("block did not queue abandon_work: %s", ran)
	}
	if payloads, _ := receivedAbandons(t, c, 0); len(payloads) != 0 {
		t.Fatalf("worker abandoned a request a caller still holds: %s", payloads)
	}
	c.action("ABANDON_USERS", 0, "-1", 0)
	if ran := string(c.action("ABANDON_STEP", 0, "", 0)); ran != "false" {
		t.Fatal("abandon_work queued twice")
	}
	c.action("ABANDON_PASS", 0, "", 0)
	payloads, responses := receivedAbandons(t, c, 0)
	want := mustRuntimeJSON(t, map[string]any{"anchor": testUUID(101).String(), "nonce": testUUID(103).String(), "realm": testUUID(100).String(), "subject": "616263",
		"claims": []map[string]any{{"number": 2, "original_hash": originalHash}}})
	if len(payloads) != 1 || !bytes.Equal(payloads[0], want) {
		t.Fatalf("ABANDON bytes %s want %s", payloads, want)
	}
	number, status, abandonProof := receivedAbandonOutcome(t, responses[0])
	if number != 2 || status != "ABANDONED" || len(abandonProof) != 64 {
		t.Fatalf("ABANDON outcome %d %s %q", number, status, abandonProof)
	}
	if flags := string(c.action("ABANDON_FLAGS", 0, "", 0)); flags != `{"abandoning":true,"gap":true,"terminal":false}` {
		t.Fatalf("ABANDONED not mapped to gap: %s", flags)
	}
	c.action("ABANDON_PASS", 0, "", 0)
	if payloads, _ := receivedAbandons(t, c, 0); len(payloads) != 1 {
		t.Fatalf("second pass sent ABANDON again: %d", len(payloads))
	}
	c.action("RETIRE", 0, "", 0)
	if floor, requests, _ := receivedState(t, c, 0); floor != 2 || requests != 0 {
		t.Fatalf("abandoned claim not retired: floor %d requests %d", floor, requests)
	}
	for cell, command := range c.commands {
		if command.Op != "GAP_RESOLVE" || command.Context.Session != r.peers[0].scope.Session {
			continue
		}
		var reply struct {
			Payload struct {
				Proof string `json:"proof_hash"`
			}
		}
		require(t, json.Unmarshal(c.responses[cell], &reply))
		if reply.Payload.Proof != abandonProof {
			t.Fatalf("GAP_RESOLVE proof %s differs from ABANDON proof %s", reply.Payload.Proof, abandonProof)
		}
	}

	c.action("FAULT", 1, "CALL 2", 0)
	c.action("REQUEST", 1, args, -5)
	if ran := string(c.action("ABANDON_STEP", 1, "", 0)); ran != "true" {
		t.Fatalf("block did not queue abandon_work: %s", ran)
	}
	payloads, responses = receivedAbandons(t, c, 1)
	if len(payloads) != 1 {
		t.Fatalf("session 1 ABANDONs %d", len(payloads))
	}
	if number, status, proof := receivedAbandonOutcome(t, responses[0]); number != 2 || status != "KEPT" || proof != "" {
		t.Fatalf("prepared claim not KEPT: %d %s %q", number, status, proof)
	}
	queries := 0
	for _, command := range c.commands {
		queries += map[bool]int{true: 1}[command.Op == "QUERY" && command.Context.Session == r.peers[1].scope.Session]
	}
	// A rerun sends nothing for the listed request (the CALL is refused at
	// admission, before any QUERY); see the test comment for what this does
	// not cover.
	c.action("REQUEST_CONTINUE", 1, "", -112)
	after := 0
	for _, command := range c.commands {
		after += map[bool]int{true: 1}[command.Op == "QUERY" && command.Context.Session == r.peers[1].scope.Session]
	}
	if after != queries {
		t.Fatalf("QUERY sent for an abandoning request: %d -> %d", queries, after)
	}
	if flags := string(c.action("ABANDON_FLAGS", 1, "", 0)); flags != `{"abandoning":true,"gap":false,"terminal":false}` {
		t.Fatalf("KEPT request changed: %s", flags)
	}
	c.action("RETAIN_VERIFY", 0, "", 0)
	c.close()
}

// TestControlledReceivedActualEffectedGapRefusal loses the reply of an admitted
// CREATE, checks that the CREATE took effect on the host and that the host
// then refuses GAP for that effected claim while the guest keeps ownership.
//
// Since oc4+pc4 (merge 942614a) the lost reply blocks the session, and a
// blocked session polls only exempt operations (DRAIN_WRITE, GAP_RESOLVE and
// the gap QUERY, dh4). The guest's plain QUERY of the claim therefore halts
// with -EHOSTDOWN at its first AGAIN, so the effect is read from the host's
// claim ledger instead (fl4). That lookup is read-only: it sends no command
// and does not change what the guest's GAP_RESOLVE sees.
func TestControlledReceivedActualEffectedGapRefusal(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.MappedProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	// The lost CALL leaves cells the host never answers; since z54 their
	// reads are a definitive IO (bh4).
	c.passReadErrors = true
	root := c.action("REQUEST", 0, "ROOT "+hex.EncodeToString(mustRuntimeJSON(t, map[string]any{})), 0)
	caps := receivedResult(t, root)
	c.action("ADOPT_ROOT_PREPARE", 0, "", 0)
	c.action("FAULT", 0, "CALL 2", 0)
	args := map[string]any{"parent": caps[0], "name": "6566666563746564", "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}
	c.action("REQUEST", 0, "CREATE "+hex.EncodeToString(mustRuntimeJSON(t, args)), -5)
	claim := receivedCallClaim(t, c, "CREATE")
	if claim.Number != 2 {
		t.Fatal("lost CALL did not retain actual CREATE original")
	}
	requireHostCreateEffect(t, r.peers[0], claim)
	// The host refuses GAP for its protected effected claim (a prepared claim
	// is Unavailable to ResolveOriginalGap) and publishes no response. The
	// exempt GAP_RESOLVE reaches the host although the session is blocked.
	// Since z54 the unanswered cell reads as a definitive EIO (lib9p: Rbridge
	// ERROR), so the guest fails at once with -EPROTO instead of waiting out
	// its -ETIMEDOUT control deadline. It retains the original
	// request/candidate; no reset follows.
	errorsBefore := c.nativeErrors
	c.action("RESOLVE_GAP", 0, "", -71)
	requireUnansweredGap(t, c, claim, errorsBefore)
	if floor, requests, unknown := receivedState(t, c, 0); floor != 0 || requests != 2 || unknown != 2 {
		t.Fatalf("effected GAP refusal lost ownership: %d/%d/%d", floor, requests, unknown)
	}
	c.action("RETAIN_VERIFY", 0, "", 0)
	c.close()
}

// receivedCallClaim returns the claim of the last actual CALL whose intent is
// opcode (a zero claim when there is none).
func receivedCallClaim(t *testing.T, c *receivedCarrier, opcode string) guestbridge.Claim {
	t.Helper()
	var claim guestbridge.Claim
	for _, command := range c.commands {
		if command.Op != "CALL" {
			continue
		}
		var call guestbridge.Call
		require(t, json.Unmarshal(command.Payload, &call))
		if call.Intent.Opcode == opcode {
			claim = call.Claim
		}
	}
	return claim
}

// requireHostCreateEffect reads claim from the peer's real host claim ledger
// (the provider the guest's commands reach) and requires a TERMINAL OK CREATE
// result with capabilities. The host runs a CALL before it writes the reply
// the guest lost, so the effect is normally already there; the short bounded
// poll only tolerates a ledger that has not reached TERMINAL yet. This
// replaces the guest QUERY a blocked session may no longer send (fl4).
func requireHostCreateEffect(t *testing.T, peer *sourceRig, claim guestbridge.Claim) {
	t.Helper()
	var state guestbridge.ClaimState
	for range 100 {
		var err error
		state, err = peer.provider.Resolve(t.Context(), peer.scope, claim)
		require(t, err)
		if state.Status == "TERMINAL" {
			break
		}
		time.Sleep(10 * time.Millisecond)
	}
	if state.Status != "TERMINAL" {
		t.Fatalf("actual admitted CREATE did not finish: %s", state.Status)
	}
	var result struct {
		Code         string
		Capabilities []guestbridge.Capability
	}
	require(t, json.Unmarshal(state.Result, &result))
	if result.Code != "OK" || len(result.Capabilities) == 0 {
		t.Fatalf("actual CREATE effect absent: %s", state.Result)
	}
}

// requireUnansweredGap requires that the guest sent GAP_RESOLVE for exactly
// claim, that the host answered none of those cells, and that the failure
// came from a definitive native read error (the carrier answered ERROR)
// after errorsBefore, not from a guest-side refusal before any send.
func requireUnansweredGap(t *testing.T, c *receivedCarrier, claim guestbridge.Claim, errorsBefore int) {
	t.Helper()
	sent := 0
	for cell, command := range c.commands {
		if command.Op != "GAP_RESOLVE" {
			continue
		}
		var call guestbridge.Call
		require(t, json.Unmarshal(command.Payload, &call))
		if call.Claim != claim {
			t.Fatalf("GAP_RESOLVE for another claim: %+v", call.Claim)
		}
		if c.responses[cell] != nil {
			t.Fatalf("host answered GAP_RESOLVE of an effected claim: %s", c.responses[cell])
		}
		sent++
	}
	if sent == 0 || c.nativeErrors == errorsBefore {
		t.Fatalf("effected GAP not refused by the host: %d sent, %d native errors", sent, c.nativeErrors-errorsBefore)
	}
}

// Full 80-cell composition of the actual C exporter with the actual Go owners.
// This mode explicitly substitutes the OS/kernel observer, as the host-only
// owner tests do. It does not execute Linux teardown or mint private Received;
// the separate dirty test exercises those production-source lifetime paths.
func TestControlledReceivedActualTerminalBackend80(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.TerminalProtocol(), 2, receivedInitialIDMap)
	c := startReceivedCarrierMode(t, r, true)
	exchange := func(session int, cell uint64, op string, payload any, ack *guestbridge.Ack) ([]byte, []byte, guestbridge.VerificationReceiptV2) {
		peer := r.peers[session]
		original := mustRuntimeJSON(t, map[string]any{"cell": cell, "context": peer.scope, "delegation": nil, "op": op, "payload": payload, "protocol": guestbridge.TerminalProtocol(), "reclaim": ack})
		response := c.action("BACKEND_EXCHANGE", session, hex.EncodeToString(original), 0)
		var receipt guestbridge.VerificationReceiptV2
		if op == "SESSION_RETIRE" {
			var err error
			receipt, err = guestbridge.VerifyMappedReceipt(peer.mappedBinding(), original, response)
			require(t, err)
		} else {
			// HELLO/blocked RESUME own transport replies, not source or terminal
			// authority: their existing closed contract has verification:null.
			var envelope struct {
				Cell         uint64
				Context      guestbridge.Context
				Op           string
				Protocol     guestbridge.Protocol
				Verification json.RawMessage
				Payload      struct{ Status string }
			}
			require(t, json.Unmarshal(response, &envelope))
			status := "ACCEPTED"
			if op == "RESUME" {
				status = "BLOCKED"
			}
			if envelope.Cell != cell || envelope.Context != peer.scope || envelope.Op != op || envelope.Protocol != guestbridge.TerminalProtocol() || string(envelope.Verification) != "null" || envelope.Payload.Status != status {
				t.Fatalf("invalid actual control reply: %s", response)
			}
		}
		return original, response, receipt
	}
	old := []terminalOldCell{}
	for i := 0; i < 80; i++ {
		op, value := "HELLO", any(guestbridge.TerminalProtocol())
		control := i >= 64
		if control {
			op = "RESUME"
			value = map[string]any{"previous_session": r.scope.Session, "ledger_hash": hashBytes([]byte("ledger")), "claims_hash": hashBytes([]byte("claims")), "domains_hash": hashBytes([]byte("domains"))}
		}
		r.cells++
		_, response, _ := exchange(0, r.cells, op, value, nil)
		nativeCell := uint64(0)
		for cell, command := range c.commands {
			if command.Context == r.scope && command.Delegation.GuestCell == r.cells {
				if nativeCell != 0 {
					t.Fatal("duplicate native original")
				}
				nativeCell = cell
			}
		}
		if nativeCell == 0 || len(c.responses[nativeCell]) == 0 {
			t.Fatal("missing actual native response")
		}
		old = append(old, terminalOldCell{guest: guestbridge.Ack{Cell: r.cells, ResponseHash: hashBytes(response)}, native: guestbridge.Ack{Cell: nativeCell, ResponseHash: hashBytes(c.responses[nativeCell])}, control: control})
	}
	type stats struct {
		Rows, Ordinary, Control, Old, Phase int
		High                                uint64
		Removed                             bool
		PackageBytes                        uint64 `json:"package_bytes"`
	}
	var before stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 0, "", 0), &before))
	if before.Rows != 80 || before.Ordinary != 64 || before.Control != 16 || before.High != 80 || before.PackageBytes == 0 {
		t.Fatalf("wrong actual exporter pre-cut: %+v", before)
	}
	request, donor, _, _ := terminalPublishCut(t, r, old, 0)
	r.cells++
	original, response, receipt := exchange(0, r.cells, "SESSION_RETIRE", request, donor)
	if receipt.Kind != "SESSION_RETIRED" || receipt.PollCount != 1 {
		t.Fatal("missing real signed terminal receipt")
	}
	var final stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 0, "", 0), &final))
	if final.Rows != 1 || final.Ordinary != 1 || final.Control != 0 || final.Old != 80 || !final.Removed || final.Phase != 3 || final.High != 81 || final.PackageBytes != before.PackageBytes {
		t.Fatalf("wrong actual exporter final owner: %+v", final)
	}
	// Exact same-live replay retains one native cell and identical full bytes.
	_, replay, _ := exchange(0, r.cells, "SESSION_RETIRE", request, donor)
	if !bytes.Equal(response, replay) {
		t.Fatal("exporter replay changed signed bytes")
	}
	_, _, _ = exchange(1, 1, "HELLO", guestbridge.TerminalProtocol(), nil)
	var peer stats
	require(t, json.Unmarshal(c.action("BACKEND_STATS", 1, "", 0), &peer))
	if peer.High != 82 || peer.Rows != 1 || peer.Ordinary != 2 || peer.Control != 0 {
		t.Fatalf("shared native allocator stopped or reset: %+v", peer)
	}
	c.close()
	state, cells := terminalInspect(t, r)
	if state.Ordinary != 2 || state.Control != 0 || state.Workspace || len(cells) != 2 || state.Terminal[r.scope.Session] != 81 {
		t.Fatalf("wrong actual C8 final cut: %+v", state)
	}
	found := false
	for _, cell := range cells {
		if cell.Header.Cell != 81 {
			continue
		}
		found = true
		if cell.Terminal == nil || cell.Terminal.Phase != "TERMINAL_RESPONDED" || cell.Terminal.OldOrdinary != 64 || cell.Terminal.OldControl != 16 || len(cell.Terminal.Donors) != 79 {
			t.Fatal("C8 lost full original manifest/class owners")
		}
	}
	if !found {
		t.Fatal("final native C8 owner missing")
	}
	_, err := guestbridge.VerifyMappedReceipt(r.mappedBinding(), original, response)
	require(t, err)
}

// Real maintained hfs_control owns every slot and private Received object here.
// Delayed native delivery creates genuine concurrent ordinary/control candidates;
// no slot, Ack, counter, observer or successful receipt is supplied by the test.
func TestControlledReceivedActualTerminalKernel80(t *testing.T) {
	r := newSourceRigDomainsUnstarted(t, guestbridge.TerminalProtocol(), 1, receivedInitialIDMap)
	// Existing signed total cap is1GiB. Select an explicit stress enrollment;
	// default256/256MiB profiles and all implementation caps remain unchanged.
	c := startReceivedCarrierBudget(t, r, false, GuestArtifactBudget{OrdinaryBytes: 768 << 20, ControlBytes: 256 << 20, Reservations: 8192, Cookies: 4096, Received: 4096})
	receivedResult(t, c.action("REQUEST", 0, "ROOT "+hex.EncodeToString(mustRuntimeJSON(t, map[string]any{})), 0))
	c.action("ADOPT_ROOT", 0, "", 0)
	c.action("ACTOR_START", 0, "", 0)
	c.action("ACQUIRE", 0, "", 0)
	var acquire json.RawMessage
	for _, command := range c.commands {
		if command.Context == r.scope && command.Op == "VIEW_ACQUIRE" {
			acquire = bytes.Clone(command.Payload)
		}
	}
	if len(acquire) == 0 {
		t.Fatal("missing real acquisition original")
	}
	reopened := false
	c.yields = func(_ uint64) {
		var ready bool
		require(t, json.Unmarshal(c.action("RELEASE_READY", 0, "", 0), &ready))
		if !ready {
			return
		}
		c.yields = nil
		for round := 0; round < 80 && !reopened; round++ {
			for _, queue := range []string{"poll", "event", "drain", "event"} {
				c.action("ACTOR_STEP", 0, queue, 0)
			}
			var state struct{ Phase string }
			require(t, json.Unmarshal(c.action("ACTOR_STATE", 0, "", 0), &state))
			reopened = state.Phase == "REOPENED"
		}
		if !reopened {
			t.Fatal("actual clean recall actors did not finish")
		}
	}
	c.action("RELEASE_DONE", 0, "", 0)
	c.yields = nil
	if !reopened {
		t.Fatal("actual lease release did not exercise recall")
	}
	// The release's original admission fills the control slots. Since task
	// q64 the controller recycles retired ticket rows: an exact admission
	// replay under a retired number still answers a rights-free RETIRED_ID
	// of its own bytes, but VIEW_QUERY of it (which no longer names an
	// original operation) is refused, so the control lane replays the
	// original VIEW_RELEASE instead of a VIEW_QUERY.
	var release json.RawMessage
	for _, command := range c.commands {
		if command.Context == r.scope && command.Op == "VIEW_RELEASE" {
			release = bytes.Clone(command.Payload)
		}
	}
	if len(release) == 0 {
		t.Fatal("missing real release original")
	}
	c.action("REPLIES_DROP", 0, "", 0)
	c.action("RETIRE", 0, "", 0)
	c.action("VIEW_RETIRE", 0, "", 0)
	requireRetiredFloor(t, c, release)
	type kernelStats struct {
		Ordinary, Control, Reserved, Unknown, Links int
		OrdinaryCredits                             int `json:"ordinary_credits"`
		ControlCredits                              int `json:"control_credits"`
		OrdinaryReceived                            int `json:"ordinary_received"`
		ControlReceived                             int `json:"control_received"`
		CapturedOrdinary                            int `json:"captured_ordinary"`
		CapturedControl                             int `json:"captured_control"`
		Verified, Pending                           bool
		ModulePuts                                  int `json:"module_puts"`
		ClientDestroys                              int `json:"client_destroys"`
		FaultModulePuts                             int `json:"fault_module_puts"`
		FaultClients                                int `json:"fault_clients"`
		Active                                      int
		ProgressParents                             bool `json:"progress_parents"`
	}
	stats := func() kernelStats {
		var s kernelStats
		require(t, json.Unmarshal(c.action("KERNEL_STATS", 0, "", 0), &s))
		return s
	}
	baseline := stats()
	issued := 1
	c.holdNativeResponses = true
	c.yields = func(_ uint64) {
		if issued == 80 {
			c.holdNativeResponses = false
			c.yields = nil
			return
		}
		reserved, op, payload := 0, "VIEW_ACQUIRE", acquire
		if issued >= 64 {
			reserved, op, payload = 1, "VIEW_RELEASE", release
		}
		issued++
		c.action("REPLAY_CONTROL", 0, fmt.Sprintf("%d %s %s", reserved, op, hex.EncodeToString(payload)), 0)
	}
	// The single-threaded consumer issues the other 79 replays nested in
	// the first one's scheduling yields (each in the previous one's), so
	// on the wall clock all 80 sends and their unwinding ran inside the
	// first request's 30 s hfs_transport deadline. That took 18.8 s at
	// load 4 and exceeded 30 s (-110) next to a kernel build (k84). A
	// kernel would run them on separate threads, so CLOCK_SPLIT_BEGIN
	// charges each nested request's service time only to itself (ml4,
	// compat.h) for the fill. Unlike k84's CLOCK_HOLD, which froze jiffies,
	// every request's own deadline stays live: a stalled request fails
	// with its own -110 instead of hanging until the go test timeout. The
	// log records the fill's real wall time.
	fillStart := time.Now()
	c.action("CLOCK_SPLIT_BEGIN", 0, "", 0)
	c.action("REPLAY_CONTROL", 0, "0 VIEW_ACQUIRE "+hex.EncodeToString(acquire), 0)
	c.action("CLOCK_SPLIT_END", 0, "", 0)
	t.Logf("kernel80 fill: %d nested control replays in %s wall time (nested service time charged per request)", issued, time.Since(fillStart).Round(time.Millisecond))
	c.yields = nil
	c.holdNativeResponses = false
	full := stats()
	if issued != 80 || full.Ordinary != 64 || full.Control != 16 || full.Reserved != 0 || full.Unknown != 0 || full.Links != baseline.Links || full.OrdinaryCredits != baseline.OrdinaryCredits || full.ControlCredits != baseline.ControlCredits || full.OrdinaryReceived != baseline.OrdinaryReceived || full.ControlReceived != baseline.ControlReceived {
		t.Fatalf("actual80-slot/private Received cut or released transient credits wrong: baseline%+v full%+v issued%d", baseline, full, issued)
	}
	c.action("ROOT_DROP", 0, "", 0)
	c.action("ARM_POST_HANDOFF_FAULT", 0, "", 0)
	c.action("DESTROY", 0, "", -116)
	transferred := stats()
	if !transferred.Verified || !transferred.Pending || transferred.Ordinary != 0 || transferred.Control != 0 || transferred.Reserved != 0 || transferred.Unknown != 0 || transferred.Links != 0 || transferred.CapturedOrdinary != 64 || transferred.CapturedControl != 16 || transferred.ProgressParents || transferred.ModulePuts != transferred.FaultModulePuts || transferred.ClientDestroys != transferred.FaultClients || transferred.Active != 1 {
		t.Fatalf("actual guest80 handoff lost retained failure owner or leaked private completion: %+v", transferred)
	}
	before := len(c.commands)
	c.action("RESTORE_POST_HANDOFF", 0, "", 0)
	c.action("DESTROY", 0, "", 0)
	if len(c.commands) != before {
		t.Fatal("post-handoff retry issued another RPC")
	}
	c.action("TOMBSTONE", 0, "", 0)
	c.close()
	state, cells := terminalInspect(t, r)
	if state.Ordinary != 1 || state.Control != 0 || len(cells) != 1 || cells[0].Terminal == nil || cells[0].Terminal.OldOrdinary != 64 || cells[0].Terminal.OldControl != 16 || len(cells[0].Terminal.Donors) != 79 {
		t.Fatal("actual guest80 handoff lost retained native manifest/classes")
	}
}

// requireRetiredFloor requires an actual RETIRED reply of the release
// ticket's own control family whose floor covers the release's number.
// Only then is the release row recycled, so its replays answer RETIRED_ID;
// a still-live row would answer PENDING/RELEASED and consumer.c's RETIRED_ID
// assert would fail without naming the cause.
func requireRetiredFloor(t *testing.T, c *receivedCarrier, release json.RawMessage) {
	t.Helper()
	var original struct{ Request guestbridge.Ticket }
	require(t, json.Unmarshal(release, &original))
	floor := uint64(0)
	for cell, command := range c.commands {
		if command.Op != "VIEW_RETIRE" || command.Context != c.rig.scope {
			continue
		}
		var cut struct{ Anchor, Nonce string }
		require(t, json.Unmarshal(command.Payload, &cut))
		var reply struct {
			Payload struct {
				Status string
				Floor  uint64
			}
		}
		if cut.Anchor != original.Request.Anchor || cut.Nonce != original.Request.Nonce || json.Unmarshal(c.responses[cell], &reply) != nil || reply.Payload.Status != "RETIRED" {
			continue
		}
		floor = max(floor, reply.Payload.Floor)
	}
	if floor < original.Request.Number {
		t.Fatalf("control family floor %d does not retire the release ticket %d: its replays would not answer RETIRED_ID", floor, original.Request.Number)
	}
}

 c.action("BACKEND_ADD", 0, hex.EncodeToString(mustRuntimeJSON(t, r.peers[16].scope)), -28)
 for i := range r.peers[:16] {
  response := c.action("BACKEND_EXCHANGE", i, hex.EncodeToString(originals[i]), 0)
  if !bytes.Equal(response, responses[i]) { t.Fatal("permanent signed bytes changed") }
  var s stats; require(t, json.Unmarshal(c.action("BACKEND_STATS", i, "", 0), &s))
  if s.Rows != 1 || s.Ordinary != 0 || s.Control != 16 || s.High != 16 || !s.Removed || s.Phase != 3 || s.PackageBytes != 88504 { t.Fatalf("permanent exporter owners evicted/refunded: %+v", s) }
 }
 if len(c.commands) != beforeCommands || beforeCommands != 16 { t.Fatal("exhaustion/replay issued another native cell") }
 c.close()
 state, cells := terminalInspect(t, r)
 if state.Control != 16 || state.Ordinary != 0 || state.Workspace || len(cells) != 16 || len(state.Terminal) != 16 { t.Fatalf("permanent C8 owners changed: %+v", state) }
 for _, cell := range cells { if cell.Terminal == nil || cell.Terminal.Phase != "TERMINAL_RESPONDED" || !cell.Control { t.Fatal("lost permanent native final") } }
}

// Exercise actual hfs_call and unchanged VFS/workqueue cleanup wrappers against
// the real exporter/host. Exhaustion must not consume a claim or native cell.
func TestControlledReceivedActualCleanupByteFunding(t *testing.T) {
	for _, protocol := range []guestbridge.Protocol{guestbridge.MappedProtocol(), guestbridge.TerminalProtocol()} {
		t.Run(fmt.Sprint(protocol.Features), func(t *testing.T) {
			r := newSourceRigDomainsUnstarted(t, protocol, 2, receivedInitialIDMap)
			c := startReceivedCarrier(t, r)
			setup := func(session int, wrappers bool) (guestbridge.Capability, guestbridge.Capability) {
				request := func(op string, args any) []byte {
					return c.action("REQUEST", session, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
				}
				root := receivedResult(t, request("ROOT", map[string]any{}))
				c.action("ADOPT_ROOT_PREPARE", session, "", 0)
				caps := receivedResult(t, request("CREATE", map[string]any{"parent": root[0], "name": fmt.Sprintf("636c65616e7570%02x", session+48), "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
				var description, path guestbridge.Capability
				for _, cap := range caps {
					if cap.Kind == "DESCRIPTION" {
						description = cap
					}
					if cap.Kind == "PATH" {
						path = cap
					}
				}
				if description.Kind == "" || path.Kind == "" {
					t.Fatal("missing actual CREATE capability")
				}
				if wrappers {
					c.action("CLEANUP_OWNERS", session, "", 0)
				} else {
					c.action("ADOPT_DESCRIPTION", session, "CREATE", 0)
				}
				c.action("REPLIES_DROP", session, "", 0)
				c.action("RETIRE", session, "", 0)
				return description, path
			}
			description, path := setup(0, false)
			operations := []struct {
				op   string
				args any
			}{
				{"DESCRIPTION_CLOSE", map[string]any{"description": description, "process": r.peers[0].credentials.Process, "close_sequence": 1}},
				{"DESCRIPTION_RELEASE", map[string]any{"description": description, "release_sequence": 1}},
				{"FORGET", map[string]any{"object": path, "count": 1}},
				{"PROCESS_EXIT", map[string]any{"process": r.peers[0].credentials.Process, "exit_sequence": 1}},
			}
			c.action("BYTE_PRESSURE", 0, "ordinary", 0)
			before := len(c.commands)
			for _, item := range operations {
				c.action("CLEANUP_REQUEST", 0, item.op+" "+hex.EncodeToString(mustRuntimeJSON(t, item.args)), -28)
			}
			if len(c.commands) != before {
				t.Fatal("unfunded cleanup emitted native bytes")
			}
			c.action("BYTE_UNPRESSURE", 0, "ordinary", 0)
			c.action("BYTE_PRESSURE", 0, "control", 0)
			for _, item := range operations {
				c.action("CLEANUP_REQUEST", 0, item.op+" "+hex.EncodeToString(mustRuntimeJSON(t, item.args)), 0)
				// The full reply remains live through the C owner/retry/retirement checks.
				// Reap it before the next full request, within the unchanged byte budget.
				c.action("REPLIES_DROP", 0, "", 0)
				c.action("RETIRE", 0, "", 0)
			}
			c.action("BYTE_UNPRESSURE", 0, "control", 0)
			setup(1, true)
			c.action("BYTE_PRESSURE", 0, "ordinary", 0)
			before = len(c.commands)
			c.action("CALLERS_UNFUNDED", 1, "", 0)
			if len(c.commands) != before {
				t.Fatal("unfunded production caller emitted native bytes")
			}
			c.action("BYTE_UNPRESSURE", 0, "ordinary", 0)
			c.action("BYTE_PRESSURE", 0, "control", 0)
			for _, item := range operations {
				c.action("CLEANUP_CALLER", 1, item.op, 0)
				c.action("RETIRE", 1, "", 0)
			}
			c.action("BYTE_UNPRESSURE", 0, "control", 0)
			seen := map[string]int{}
			for _, command := range c.commands {
				if command.Context != r.scope || command.Op != "CALL" {
					continue
				}
				var call guestbridge.Call
				require(t, json.Unmarshal(command.Payload, &call))
				for _, item := range operations {
					if call.Intent.Opcode == item.op {
						seen[item.op]++
						if command.Delegation.GuestOp != "CALL" || !bytes.Equal(call.Intent.Args, mustRuntimeJSON(t, item.args)) || !bytes.Equal(mustRuntimeJSON(t, call.Credentials), mustRuntimeJSON(t, r.peers[0].credentials)) {
							t.Fatalf("cleanup changed original wire/credentials: %s", command.Payload)
						}
					}
				}
			}
			for _, item := range operations {
				if seen[item.op] != 1 {
					t.Fatalf("cleanup %s executed %d times", item.op, seen[item.op])
				}
			}
			c.close()
		})
	}
}

// packageRig is the 6m4 fixture: session 1 holds its own ROOT grant but no
// retained request (its ROOT is retired at once), session 0 adopts ROOT and
// a CREATEd file and keeps both requests retained but unborrowed, so only
// session 0 has anything to retire. retires counts host RETIRE exchanges.
type packageRig struct {
	c           *receivedCarrier
	path, root1 string
}

func newPackageRig(t *testing.T, protocol guestbridge.Protocol) *packageRig {
	r := newSourceRigDomainsUnstarted(t, protocol, 2, receivedInitialIDMap)
	c := startReceivedCarrier(t, r)
	request := func(session int, op string, args any) []byte {
		return c.action("REQUEST", session, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
	}
	root1 := receivedResult(t, request(1, "ROOT", map[string]any{}))
	c.action("ADOPT_ROOT_PREPARE", 1, "", 0)
	c.action("REPLIES_DROP", 1, "", 0)
	c.action("RETIRE", 1, "", 0)
	root := receivedResult(t, request(0, "ROOT", map[string]any{}))
	c.action("ADOPT_ROOT_PREPARE", 0, "", 0)
	caps := receivedResult(t, request(0, "CREATE", map[string]any{"parent": root[0], "name": "7061636b616765", "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
	var path guestbridge.Capability
	for _, cap := range caps {
		if cap.Kind == "PATH" {
			path = cap
		}
	}
	if path.Kind == "" {
		t.Fatal("missing actual CREATE PATH capability")
	}
	c.action("ADOPT_DESCRIPTION", 0, "CREATE", 0)
	c.action("REPLIES_DROP", 0, "", 0)
	return &packageRig{c: c,
		path:  hex.EncodeToString(mustRuntimeJSON(t, map[string]any{"object": path})),
		root1: hex.EncodeToString(mustRuntimeJSON(t, map[string]any{"object": root1[0]}))}
}

func (p *packageRig) retires() int {
	n := 0
	for _, command := range p.c.commands {
		if command.Op == "RETIRE" {
			n++
		}
	}
	return n
}

// expect runs one PACKAGE_* action and checks its RETIRE count; a refusal
// must also emit no native command at all.
func (p *packageRig) expect(t *testing.T, verb string, session int, args string, want, retired int) {
	t.Helper()
	before, commands := p.retires(), len(p.c.commands)
	p.c.action(verb, session, args, want)
	if p.retires() != before+retired {
		t.Fatalf("%s session%d: %d RETIRE exchanges, want %d", verb, session, p.retires()-before, retired)
	}
	if want != 0 && len(p.c.commands) != commands {
		t.Fatalf("refused %s session%d emitted native bytes", verb, session)
	}
}

// logged requires the consumer's stderr (pr_info_ratelimited in compat.h)
// to hold hfs_retire_shared_note's line with this tail (km4, zq3-r22 R22-1:
// the guest log must show whether the retire-and-retry path ran and what it
// freed). The consumer's stderr is copied by its own goroutine, so the line
// may arrive after the action's stdout reply: wait up to 10 s for it.
func (p *packageRig) logged(t *testing.T, tail string) {
	t.Helper()
	want := "retired under charge pressure: " + tail
	for deadline := time.Now().Add(10 * time.Second); !strings.Contains(p.c.diagnostics.String(), want); {
		if time.Now().After(deadline) {
			t.Fatalf("missing retire-and-retry log line %q in:\n%s", tail, p.c.diagnostics.String())
		}
		time.Sleep(10 * time.Millisecond)
	}
}

// 6m4 (zq3-r21 R21-1): a retained request keeps its full completion package
// until retirement, and the ordinary bytes are global across mounts. When the
// partition still funds the next request's transport reservation but not its
// package, the actual hfs_call must retire and retry instead of failing
// ENOSPC locally: its own session's requests, and (r21's m1, which had
// nothing to retire) another session's. With nothing retirable anywhere (the
// oldest request's result still held) it still refuses without a claim/cell.
func TestControlledReceivedActualPackageRetirement(t *testing.T) {
	for _, protocol := range []guestbridge.Protocol{guestbridge.MappedProtocol(), guestbridge.TerminalProtocol()} {
		t.Run(fmt.Sprint(protocol.Features), func(t *testing.T) {
			p := newPackageRig(t, protocol)
			// Session 1 has nothing to retire: session 0's ROOT/CREATE must go.
			p.expect(t, "PACKAGE_REQUEST", 1, p.root1, 0, 1)
			// km4: the round is logged with what it freed.
			// 9n4: the line names the charge's byte lane.
			p.logged(t, "own retire -11 freed 0 requests, 1 of 1 other sessions freed 2, ordinary lane; retrying the charge once")
			// Session 1's GETATTR result stays held; session 0 has none left.
			p.expect(t, "PACKAGE_REQUEST", 1, p.root1, -28, 0)
			p.logged(t, "own retire -11 freed 0 requests, 0 of 1 other sessions freed 0, ordinary lane; nothing freed, the charge fails")
			// Session 0's own retirable GETATTR, same session.
			p.c.action("REQUEST", 0, "GETATTR "+p.path, 0)
			p.c.action("REPLIES_DROP", 0, "", 0)
			p.expect(t, "PACKAGE_REQUEST", 0, p.path, 0, 1)
			p.expect(t, "PACKAGE_REQUEST", 0, p.path, -28, 0)
			p.c.action("RETIRE", 0, "", 0)
			p.c.action("RETIRE", 1, "", 0)
			p.c.close()
		})
	}
}

// 9n4 (zq3-r23 R23-1): two charges on m0 failed together. The first round

// GETATTR charge fails, and a concurrent actual hfs_retire on session 0

// ROOT/CREATE. The caller must retry the charge (exactly one RETIRE, the

		})

// x84 review (zq3-r10 R10-1): at process exit PROCESS_EXIT normally reaches
// the host first; the exiting process's own DESCRIPTION_RELEASE and a FORGET
// queued later (hfs_evict_inode) then carry the credentials of an exited
// process. Both must still succeed through the unchanged production callers,
// without blocking the session.
//
// REQUIRES host task 194 (ALLOCATE accepts an exited issuer for cleanup
// intents). Without it the release's ALLOCATE is refused Stale by
// admitCredentials, hfs_release blocks the session and this test fails.
func TestControlledReceivedActualExitThenCleanup(t *testing.T) {
	for _, protocol := range []guestbridge.Protocol{guestbridge.MappedProtocol(), guestbridge.TerminalProtocol()} {
		t.Run(fmt.Sprint(protocol.Features), func(t *testing.T) {
			r := newSourceRigDomainsUnstarted(t, protocol, 2, receivedInitialIDMap)
			c := startReceivedCarrier(t, r)
			request := func(op string, args any) []byte {
				return c.action("REQUEST", 0, op+" "+hex.EncodeToString(mustRuntimeJSON(t, args)), 0)
			}
			root := receivedResult(t, request("ROOT", map[string]any{}))
			c.action("ADOPT_ROOT_PREPARE", 0, "", 0)
			caps := receivedResult(t, request("CREATE", map[string]any{"parent": root[0], "name": "657869742d636c65616e7570", "mode": 0660, "umask": 0, "flags": map[string]any{"access": "READWRITE", "append": false, "create": true, "exclusive": true, "sync": "ORDINARY", "truncate": false}}))
			kinds := map[string]bool{}
			for _, cap := range caps {
				kinds[cap.Kind] = true
			}
			if !kinds["DESCRIPTION"] || !kinds["PATH"] {
				t.Fatal("missing actual CREATE capability")
			}
			c.action("CLEANUP_OWNERS", 0, "", 0)
			c.action("REPLIES_DROP", 0, "", 0)

			}

		})
	}
