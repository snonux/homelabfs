package guestbridge

import (
	"os"
	"testing"
)

func TestVZ3ExportActualHostROOT(t *testing.T) {
	f := newBridge(t)
	f.hello(t)
	cred := f.cred()
	_, result := f.call(t, cred, "ROOT", map[string]any{})
	raw, err := marshal(map[string]any{"context": f.c, "credentials": cred, "result": result})
	if err != nil {
		t.Fatal(err)
	}
	if err = os.WriteFile(os.Getenv("HFS_ROOT_FIXTURE_OUTPUT"), append(raw, '\n'), 0600); err != nil {
		t.Fatal(err)
	}
}
